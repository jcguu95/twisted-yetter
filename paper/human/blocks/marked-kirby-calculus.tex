\subsection{Kirby calculus for closed oriented marked four-manifolds}
\label{subsec:marked-kirby-calculus}

We now specify the object on which marked Kirby calculus acts.

\begin{definition}[Marked Kirby datum]
\label{def:marked-special-kirby-presentation}
\label{def:handle-homology-marking}
An \(H_2\)-marked four-manifold is a pair \((X,\alpha)\), where \(X\) is a
closed, connected, oriented smooth four-manifold and
\(\alpha\in H_2(X;\mathbb Z)\).

Let \(\mathcal H\) be a special Kirby presentation of \(X\), and set
\[
  H^2(\mathcal H)
  :=
  H^2\!\left(\operatorname{Hom}
    (C_*(\mathcal H),\mathbb Z)\right).
\]
The handle filtration and cellular-to-singular comparison give a natural
isomorphism
\[
  \kappa_{\mathcal H}\colon
  H^2(\mathcal H)\xrightarrow{\ \cong\ }H^2(X;\mathbb Z).
\]
A \emph{marked Kirby datum} for \((X,\alpha)\) is a pair
\((\mathcal H,\xi)\), where \(\xi\in H^2(\mathcal H)\) satisfies
\[
  \kappa_{\mathcal H}(\xi)=\operatorname{PD}_X(\alpha).
  \tag{3.2}\label{eq:marked-presentation}
\]
Equivalently,
\(\xi=\kappa_{\mathcal H}^{-1}(\operatorname{PD}_X(\alpha))\), so the
marking is uniquely determined once \(\mathcal H\) and \(\alpha\) are
fixed.  We also call \((\mathcal H,\xi)\) a
\emph{marked special Kirby presentation}.
\end{definition}

\begin{definition}[Transport under a Kirby--Cerf move]
\label{def:marked-kirby-move}
Let \(m\colon\mathcal H\to\mathcal H'\) be an elementary Kirby--Cerf move
in \cref{def:kirby-move}.  Its standard local model
determines a relative isotopy class of orientation-preserving
diffeomorphisms
\[
  \phi_m\colon X_{\mathcal H}\longrightarrow X_{\mathcal H'}.
\]
Define
\[
  T_m
  :=
  \kappa_{\mathcal H'}^{-1}
  \circ(\phi_m^{-1})^*
  \circ\kappa_{\mathcal H}
  \colon H^2(\mathcal H)\longrightarrow H^2(\mathcal H').
  \tag{3.3}\label{eq:marking-transport}
\]
The result of applying \(m\) to \((\mathcal H,\xi)\) is
\[
  m_*(\mathcal H,\xi):=(\mathcal H',T_m\xi).
\]
This is independent of the representative of the relative isotopy class
of \(\phi_m\).  A move equipped with this transport is called a
\emph{marked Kirby--Cerf move}.
\end{definition}

\begin{definition}[Handle cocycle]
\label{def:handle-cocycle}
Let \((\mathcal H,\xi)\) be a marked Kirby datum.  A \emph{handle cocycle}
for \((\mathcal H,\xi)\) is a cocycle
\[
  a\in\operatorname{Hom}(C_2(\mathcal H),\mathbb Z)
\]
representing \(\xi\).  If \(e_j\) is the oriented \(2\)-handle generator
corresponding to the \(j\)th framed component, write \(a_j:=a(e_j)\).
\end{definition}

\begin{proposition}[Cocycle transport under Kirby--Cerf moves]
\label{prop:marking-coordinate-transport}
Let \(m\colon\mathcal H\to\mathcal H'\) be an elementary Kirby--Cerf move,
let \(\xi\in H^2(\mathcal H)\), and let \(a\) be a handle cocycle
representing \(\xi\).  Thus \(T_m\xi\) is the transported cohomology class
and \(a\) is the integer cocycle whose coordinates are used by the
evaluator.  The class \(T_m\xi\) has a representative \(a'\) with the
following coordinates.
\begin{enumerate}[label=(\roman*),ref=\roman*]
  \item For framed isotopy, a \(1/1\) slide, a \(2/1\) slide, or a
  \(3/3\) slide, the \(2\)-handle generators are canonically identified and
  \(a'=a\).

  \item For a \(2/2\) slide, let \(e_x\) be the moving generator,
  \(e_y\) the stationary generator, and let
  \(\varepsilon\in\{\pm1\}\) be fixed by the oriented handle-slide
  convention.  Then
  \[
    a_x'=a_x+\varepsilon a_y,
    \qquad
    a_y'=a_y,
    \tag{3.4}\label{eq:two-two-marking-transport}
  \]
  while every uninvolved coordinate is unchanged.

  \item A \(1\)--\(2\) birth admits \(a'=a\oplus0\); every representative
  of the transported class is cohomologous to one whose new \(2\)-handle
  coordinate is zero.

  \item For a \(2\)--\(3\) birth, the cocycle equation forces the new
  \(2\)-handle coordinate to be zero, so again \(a'=a\oplus0\).
\end{enumerate}
The death formulas are the inverses of the corresponding birth formulas.
\end{proposition}

\begin{proof}
The handle complex is the cellular chain complex of the handle filtration,
and cellular comparison is natural.  It is therefore enough to use the
standard cellular map for each local move
\cite[Sections~4.1--4.2]{DurstGeigesKegel2019}.

Isotopy, \(1/1\), \(2/1\), and \(3/3\) moves act as the identity on
\(C_2\).  For a \(2/2\) slide, the target-to-source basis change sends the
new moving generator to \(e_x+\varepsilon e_y\) and fixes the other
generators.  Precomposing \(a\) with this basis change gives
\[
  a_x'=a(e_x+\varepsilon e_y)=a_x+\varepsilon a_y.
\]

A \(1\)--\(2\) birth adds the acyclic complex
\[
  0\longrightarrow\mathbb Z
  \xrightarrow{\ \pm1\ }\mathbb Z\longrightarrow0
\]
in degrees \(2\) and \(1\).  Its new degree-two cochain is a coboundary, so
that coordinate may be chosen to be zero.  A \(2\)--\(3\) birth adds the
same acyclic complex in degrees \(3\) and \(2\); the degree-two cocycle
equation now forces the new coordinate to vanish.  Reversing these chain
equivalences gives the death formulas.
\end{proof}

\begin{proposition}[Kirby--Cerf moves preserve marked data]
\label{prop:kirby-move-preserves-marked-data}
Let \(m\colon\mathcal H\to\mathcal H'\) be an elementary Kirby--Cerf move,
and let \(\phi_m\colon X_{\mathcal H}\to X_{\mathcal H'}\) be the
orientation-preserving diffeomorphism represented by its standard local
model, as in \cref{def:marked-kirby-move}.  If
\((\mathcal H,\xi)\) is a marked Kirby datum for \((X,\alpha)\), then
\[
  m_*(\mathcal H,\xi)
  =
  (\mathcal H',T_m\xi)
\]
is a marked Kirby datum for
\[
  \bigl(X_{\mathcal H'},(\phi_m)_*\alpha\bigr).
\]
In particular, when the move is performed on a fixed manifold using its
standard identification, it preserves the original marking \(\alpha\).
\end{proposition}

\begin{proof}
By \eqref{eq:marking-transport} and naturality of Poincar\'e duality,
\[
  \begin{aligned}
  \kappa_{\mathcal H'}(T_m\xi)
  &=(\phi_m^{-1})^*\kappa_{\mathcal H}(\xi)\\
  &=(\phi_m^{-1})^*\operatorname{PD}_X(\alpha)\\
  &=\operatorname{PD}_{X_{\mathcal H'}}
      \bigl((\phi_m)_*\alpha\bigr).
  \end{aligned}
\]
This is exactly the defining condition \eqref{eq:marked-presentation}.
\end{proof}

\begin{theorem}[Marked Kirby--Cerf completeness]
\label{prop:marked-kirby-cerf}
Let
\[
  f\colon(X,\alpha)\longrightarrow(Y,\alpha')
\]
be an orientation-preserving diffeomorphism of \(H_2\)-marked closed
four-manifolds; thus \(f_*\alpha=\alpha'\).  Let
\((\mathcal H,\xi)\) and \((\mathcal H',\xi')\) be marked Kirby data for
\((X,\alpha)\) and \((Y,\alpha')\), respectively.  Then there is a finite
movie of marked Kirby--Cerf moves from \((\mathcal H,\xi)\) to the pullback of
\((\mathcal H',\xi')\) on \(X\).  Following that movie by the tautological
endpoint identification induced by \(f\) represents the prescribed isotopy
class of the marked diffeomorphism \(f\).
\end{theorem}

\begin{proof}
Apply ordinary Kirby--Cerf completeness,
\cref{prop:special-handle-movie-completeness}, to obtain a movie on the
fixed manifold \(X\),
\[
  \mathcal H=\mathcal H_0
  \xrightarrow{m_1}\mathcal H_1
  \longrightarrow\cdots\longrightarrow
  \mathcal H_N=f^*\mathcal H',
\]
whose realized diffeomorphism is isotopic to
\(\operatorname{id}_X\).  Starting with \(\xi_0=\xi\), define recursively
\[
  \xi_r:=T_{m_r}\xi_{r-1}.
\]
By \cref{prop:kirby-move-preserves-marked-data}, every
\((\mathcal H_r,\xi_r)\) is a marked Kirby datum for the corresponding
transport of \((X,\alpha)\).  Since the composite diffeomorphism of the
movie is isotopic to \(\operatorname{id}_X\), the endpoint
\((f^*\mathcal H',\xi_N)\) is a marked Kirby datum for \((X,\alpha)\).

There is only one marking on the fixed presentation \(f^*\mathcal H'\)
that satisfies \eqref{eq:marked-presentation}.  Moreover,
\[
  f^*\operatorname{PD}_Y(\alpha')
  =f^*\operatorname{PD}_Y(f_*\alpha)
  =\operatorname{PD}_X(\alpha).
\]
Hence \(\xi_N\) is precisely the pullback of \(\xi'\).  Finally,
\cref{prop:special-handle-movie-completeness} says that composing the
fixed-manifold movie with the endpoint identification induced by \(f\)
represents the prescribed isotopy class of \(f\).  The same composite
therefore represents the prescribed marked diffeomorphism.
\end{proof}
