\subsection{Kirby calculus for closed oriented marked 4-manifolds}
\label{subsec:marked-kirby-calculus}

We now specify the object on which marked Kirby calculus acts.

\begin{definition}[Marked \(4\)-manifold and marked diffeomorphism]
\label{def:marked-four-manifold}
An \(H_2\)-marked 4-manifold is a pair \((X,\alpha)\), where \(X\) is a
closed, connected, oriented smooth 4-manifold and
\(\alpha\in H_2(X;\mathbb Z)\).
A \emph{marked diffeomorphism}
\[
  f\colon(X,\alpha)\longrightarrow(Y,\beta)
\]
is an orientation-preserving diffeomorphism \(f\colon X\to Y\) satisfying
\(f_*\alpha=\beta\).
\end{definition}

\begin{definition}[Marked Kirby datum]
\label{def:handle-homology-marking}
A \emph{marked Kirby datum} is a pair \((\mathcal H,\xi)\), where
\(\mathcal H\) is a Kirby datum and
\[
  \xi\in H^2(\mathcal H).
\]
\end{definition}

\begin{definition}[Realization of a marked Kirby datum]
\label{def:marked-kirby-realization}
The realization of \((\mathcal H,\xi)\) is the \(H_2\)-marked 4-manifold
\[
  |(\mathcal H,\xi)|
  :=\bigl(|\mathcal H|,\alpha_{\mathcal H,\xi}\bigr),
  \qquad
  \alpha_{\mathcal H,\xi}
  :=\operatorname{PD}_{|\mathcal H|}^{-1}
      \bigl(\kappa_{\mathcal H}(\xi)\bigr).
\]
\end{definition}

\begin{definition}[Marked Kirby presentation]
\label{def:marked-kirby-presentation}
A \emph{marked Kirby presentation} of \((X,\alpha)\) is a marked Kirby
datum \((\mathcal H,\xi)\) together with a marked diffeomorphism
\[
  \rho\colon |(\mathcal H,\xi)|\longrightarrow(X,\alpha).
\]
Equivalently, \(\rho\colon|\mathcal H|\to X\) is orientation preserving and
\[
  \kappa_{\mathcal H}(\xi)=\rho^*\operatorname{PD}_X(\alpha).
  \tag{3.2}\label{eq:marked-presentation}
\]
Thus \(\xi\) is uniquely determined by an unmarked presentation
\((\mathcal H,\rho)\) and the class \(\alpha\).
\end{definition}

\begin{definition}[Transport under a Kirby--Cerf move]
\label{def:marked-kirby-move}
Let \(m\colon\mathcal H\to\mathcal H'\) be an elementary Kirby--Cerf move
in \cref{def:kirby-move}.  By
\cref{prop:elementary-move-realization}, its local model determines a
relative isotopy class of orientation-preserving
diffeomorphisms
\[
  |m|\colon |\mathcal H|\longrightarrow|\mathcal H'|.
\]
Define
\[
  T_m
  :=
  \kappa_{\mathcal H'}^{-1}
  \circ(|m|^{-1})^*
  \circ\kappa_{\mathcal H}
  \colon H^2(\mathcal H)\longrightarrow H^2(\mathcal H').
  \tag{3.3}\label{eq:marking-transport}
\]
The result of applying \(m\) to a marked Kirby datum is
\[
  m_*(\mathcal H,\xi):=(\mathcal H',T_m\xi).
\]
This is independent of the representative of the relative isotopy class
of \(|m|\).  A move equipped with this transport is called a
\emph{marked Kirby--Cerf move}.
\end{definition}

\begin{definition}[Handle cocycle]
\label{def:handle-cocycle}
Let \((\mathcal H,\xi)\) be a marked Kirby datum.  A
\emph{handle cocycle} for \((\mathcal H,\xi)\) is a cocycle
\[
  a\in\operatorname{Hom}(C_2(\mathcal H),\mathbb Z)
\]
representing \(\xi\).  If \(e_j\) is the oriented \(2\)-handle generator
corresponding to the \(j\)th framed component, write \(a_j:=a(e_j)\).
\end{definition}

\begin{proposition}[Cocycle transport under Kirby--Cerf moves]
\label{prop:marking-coordinate-transport}
Let \(m\colon\mathcal H\to\mathcal H'\) be an elementary Kirby--Cerf move,
let \(\xi\in H^2(\mathcal H)\), and let \(a\) be a handle cocycle
representing \(\xi\).  Thus \(T_m\xi\) is the transported cohomology class
and \(a\) is the integer cocycle whose coordinates are used by the
evaluator.  The class \(T_m\xi\) has a representative \(a'\) with the
following coordinates.
\begin{enumerate}[label=(\roman*),ref=\roman*]
  \item For framed isotopy, a \(1/1\) slide, a \(2/1\) passage, a
  \(3/3\) slide, a \(0\)--\(1\) birth or death, or a \(3\)--\(4\) birth or
  death, the \(2\)-handle generators are canonically identified and
  \(a'=a\).

  \item For a \(2/2\) slide, let \(e_x\) be the moving generator,
  \(e_y\) the stationary generator, and let
  \(\varepsilon\in\{\pm1\}\) be fixed by the oriented handle-slide
  convention.  Then
  \[
    a_x'=a_x+\varepsilon a_y,
    \qquad
    a_y'=a_y,
    \tag{3.4}\label{eq:two-two-marking-transport}
  \]
  while every uninvolved coordinate is unchanged.

  \item A \(1\)--\(2\) birth admits \(a'=a\oplus0\); every representative
  of the transported class is cohomologous to one whose new \(2\)-handle
  coordinate is zero.

  \item For a \(2\)--\(3\) birth, the cocycle equation forces the new
  \(2\)-handle coordinate to be zero, so again \(a'=a\oplus0\).
\end{enumerate}
The death formulas are the inverses of the corresponding birth formulas.
\end{proposition}

\begin{proof}
The handle complex is the cellular chain complex of the handle filtration,
and cellular comparison is natural.  It is therefore enough to use the
standard cellular map for each local move
\cite[Sections~4.1--4.2]{DurstGeigesKegel2019}.

Isotopy, \(1/1\), \(2/1\), \(3/3\), \(0\)--\(1\), and \(3\)--\(4\) moves
act as the identity on \(C_2\).  For a \(2/2\) slide, the target-to-source
basis change sends the
new moving generator to \(e_x+\varepsilon e_y\) and fixes the other
generators.  Precomposing \(a\) with this basis change gives
\[
  a_x'=a(e_x+\varepsilon e_y)=a_x+\varepsilon a_y.
\]

A \(1\)--\(2\) birth adds the acyclic complex
\[
  0\longrightarrow\mathbb Z
  \xrightarrow{\ \pm1\ }\mathbb Z\longrightarrow0
\]
in degrees \(2\) and \(1\).  Its new degree-two cochain is a coboundary, so
that coordinate may be chosen to be zero.  A \(2\)--\(3\) birth adds the
same acyclic complex in degrees \(3\) and \(2\); the degree-two cocycle
equation now forces the new coordinate to vanish.  Reversing these chain
equivalences gives the death formulas.
\end{proof}

\begin{proposition}[Kirby--Cerf moves preserve marked data]
\label{prop:kirby-move-preserves-marked-data}
Let \(m\colon\mathcal H\to\mathcal H'\) be an elementary Kirby--Cerf move,
and let \((\mathcal H,\xi)\) be a marked Kirby datum.  Then
\[
  |m|\colon |(\mathcal H,\xi)|
  \longrightarrow |(\mathcal H',T_m\xi)|
\]
is a marked diffeomorphism.  Consequently, if
\(\rho\colon|(\mathcal H,\xi)|\to(X,\alpha)\) is a marked Kirby
presentation, then
\[
  \rho_m:=\rho\circ|m|^{-1}
  \colon|(\mathcal H',T_m\xi)|\longrightarrow(X,\alpha)
\]
is a marked Kirby presentation of the same marked 4-manifold.
\end{proposition}

\begin{proof}
By \eqref{eq:marking-transport} and naturality of Poincar\'e duality,
\[
  \begin{aligned}
  \operatorname{PD}_{|\mathcal H'|}
    \bigl(|m|_*\alpha_{\mathcal H,\xi}\bigr)
  &=(|m|^{-1})^*\operatorname{PD}_{|\mathcal H|}
    (\alpha_{\mathcal H,\xi})\\
  &=(|m|^{-1})^*\kappa_{\mathcal H}(\xi)\\
  &=\kappa_{\mathcal H'}(T_m\xi)\\
  &=\operatorname{PD}_{|\mathcal H'|}
    (\alpha_{\mathcal H',T_m\xi}).
  \end{aligned}
\]
Since Poincar\'e duality is an isomorphism,
\(|m|_*\alpha_{\mathcal H,\xi}=\alpha_{\mathcal H',T_m\xi}\).  The final
assertion follows by composition of marked diffeomorphisms.
\end{proof}

\begin{theorem}[Marked Kirby--Cerf completeness]
\label{prop:marked-kirby-cerf}
Let
\[
  f\colon(X,\alpha)\longrightarrow(Y,\alpha')
\]
be a marked diffeomorphism.  Let
\(\rho\colon|(\mathcal H,\xi)|\to(X,\alpha)\) and
\(\rho'\colon|(\mathcal H',\xi')|\to(Y,\alpha')\) be marked Kirby
presentations.  Then there is a finite
movie
\[
  \mathcal H=\mathcal H_0
  \xrightarrow{m_1}\mathcal H_1
  \longrightarrow\cdots\longrightarrow
  \mathcal H_N=\mathcal H'
\]
such that, if \(\mu=|m_N|\circ\cdots\circ|m_1|\), then
\[
  \rho'\circ\mu\simeq f\circ\rho,
  \tag{3.6}\label{eq:marked-cerf-realizes-diffeomorphism}
\]
and successive transport by the maps \(T_{m_i}\) carries \(\xi\) to
\(\xi'\).
If both presentations are one-ended, the movie may be chosen one-ended
throughout; equivalently, it uses the nine move types other than
\(0\)--\(1\) and \(3\)--\(4\) birth/death.
\end{theorem}

\begin{proof}
Apply ordinary Kirby--Cerf completeness,
\cref{prop:handle-movie-completeness}, to obtain the displayed movie and
\eqref{eq:marked-cerf-realizes-diffeomorphism}.  Starting with
\(\xi_0=\xi\), define recursively
\[
  \xi_r:=T_{m_r}\xi_{r-1}.
\]
By \cref{prop:kirby-move-preserves-marked-data}, every stage is a marked
Kirby datum and the composite realization identifies its marking with the
source marking.  At the endpoint it gives the marked presentation
\(\rho\circ\mu^{-1}\colon|(\mathcal H',\xi_N)|\to(X,\alpha)\).  The isotopy
\eqref{eq:marked-cerf-realizes-diffeomorphism} gives
\(\rho\circ\mu^{-1}\simeq f^{-1}\circ\rho'\).  Naturality of Poincar\'e
duality gives
\[
  (f^{-1})^*\operatorname{PD}_X(\alpha)
  =\operatorname{PD}_Y(\alpha').
\]
It follows that \(\xi_N\) and \(\xi'\) have the same image under
\(\kappa_{\mathcal H'}\).  Since \(\kappa_{\mathcal H'}\) is an
isomorphism, \(\xi_N=\xi'\).
When both endpoints are one-ended, use the connected Cerf refinement in
\cref{cor:one-ended-diffeomorphism-realization}; the same
marking-transport argument applies at every stage.
\end{proof}
