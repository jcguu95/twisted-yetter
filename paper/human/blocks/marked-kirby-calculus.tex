\subsubsection{Kirby calculus for closed oriented marked 4-manifolds}
\label{subsec:marked-kirby-calculus}

We now specify the object on which marked Kirby calculus acts.

\begin{definition}[Marked \(4\)-manifold and marked diffeomorphism]
\label{def:marked-four-manifold}
An \(H_2\)-marked 4-manifold is a pair \((X,\alpha)\), where \(X\) is a
closed, connected, oriented smooth 4-manifold and
\(\alpha\in H_2(X;\mathbb Z)\).
A \emph{marked diffeomorphism}
\[
  f\colon(X,\alpha)\longrightarrow(Y,\beta)
\]
is an orientation-preserving diffeomorphism \(f\colon X\to Y\) satisfying
\(f_*\alpha=\beta\).
\end{definition}

\begin{definition}[Marked Kirby datum]
\label{def:handle-homology-marking}
A \emph{marked Kirby datum} is a pair \((\mathcal H,\beta)\), where
\(\mathcal H\) is a Kirby datum and
\[
  \beta\in H_2(|\mathcal H|;\mathbb Z).
\]
\end{definition}

\begin{definition}[Cohomological marking and handle cocycle]
\label{def:handle-cocycle}
Let \((\mathcal H,\beta)\) be a marked Kirby datum.  Dualizing the handle
chain complex of \cref{def:derived-handle-chain-complex} gives
\[
  C^k(\mathcal H):=\operatorname{Hom}(C_k(\mathcal H),\mathbb Z),
  \qquad
  \delta^k(\omega):=\omega\circ\partial_{k+1},
\]
and
\[
  H^2(\mathcal H):=
  \ker(\delta^2)/\operatorname{im}(\delta^1).
\]
Let
\[
  \kappa_{\mathcal H}\colon
  H^2(\mathcal H)\xrightarrow{\ \cong\ }
  H^2(|\mathcal H|;\mathbb Z)
\]
be the cellular comparison isomorphism.  The homology marking \(\beta\)
therefore determines the cohomology class
\[
  \xi_{\mathcal H,\beta}
  :=\kappa_{\mathcal H}^{-1}
    \bigl(\operatorname{PD}_{|\mathcal H|}(\beta)\bigr)
  \in H^2(\mathcal H).
\]
A \emph{handle cocycle} for \((\mathcal H,\beta)\) is a cocycle
\(\omega\in C^2(\mathcal H)\) representing
\(\xi_{\mathcal H,\beta}\).  If \(e_j\) is the oriented \(2\)-handle
generator corresponding to the \(j\)th framed component, write
\[
  \omega_j:=\omega(e_j)\in\mathbb Z.
\]
Thus \(\beta\) is the intrinsic marking, \(\xi_{\mathcal H,\beta}\) is
canonically derived from it, and \(\omega\) is a choice of representative
used in the evaluator; \(\omega\) is not additional marked data.
\end{definition}

\begin{definition}[Realization of a marked Kirby datum]
\label{def:marked-kirby-realization}
The realization of \((\mathcal H,\beta)\) is the \(H_2\)-marked 4-manifold
\[
  |(\mathcal H,\beta)|:=\bigl(|\mathcal H|,\beta\bigr).
\]
\end{definition}

\begin{definition}[Marked Kirby presentation]
\label{def:marked-kirby-presentation}
A \emph{marked Kirby presentation} of \((X,\alpha)\) is a marked Kirby
datum \((\mathcal H,\beta)\) together with a marked diffeomorphism
\[
  \rho\colon |(\mathcal H,\beta)|\longrightarrow(X,\alpha).
\]
Equivalently, \(\rho\colon|\mathcal H|\to X\) is orientation preserving and
\begin{equation}
  \rho_*\beta=\alpha.
  \label{eq:marked-presentation}
\end{equation}
Thus \(\beta\) is uniquely determined by an unmarked presentation
\((\mathcal H,\rho)\) and the class \(\alpha\).
\end{definition}

\begin{definition}[Marked Kirby--Cerf move]
\label{def:marked-kirby-move}
Let \(m\colon\mathcal H\to\mathcal H'\) be an elementary Kirby--Cerf move
in \cref{def:kirby-move}.  By
\cref{prop:elementary-move-realization}, its local model determines a
relative isotopy class of orientation-preserving
diffeomorphisms
\[
  |m|\colon |\mathcal H|\longrightarrow|\mathcal H'|.
\]
The marked move determined by \(m\) is
\[
  (\mathcal H,\beta)\longmapsto
  \bigl(\mathcal H',|m|_*\beta\bigr).
\]
By definition, its realization
\[
  |m|\colon|(\mathcal H,\beta)|
  \longrightarrow|(\mathcal H',|m|_*\beta)|
\]
is a marked diffeomorphism.
\end{definition}

\begin{proposition}[Cocycle coordinates under Kirby--Cerf moves]
\label{prop:marking-coordinate-transport}
Let \(m\colon(\mathcal H,\beta)\to
(\mathcal H',|m|_*\beta)\) be a marked elementary move, and let \(\omega\)
be a handle cocycle for \((\mathcal H,\beta)\).  The target class has a
representative \(\omega'\) with the following coordinates.
\begin{enumerate}[label=(\roman*),ref=\roman*]
  \item For level isotopy, a \(1/1\) slide, a \(3/3\) slide, a
  \(0\)--\(1\) birth or death, or a \(3\)--\(4\) birth or death, the
  \(2\)-handle generators are canonically identified and
  \(\omega'=\omega\).  This includes level isotopies which, in dotted
  notation, are drawn by passing a strand through a dotted circle.

  \item For a \(2/2\) slide, let \(e_x\) be the moving generator,
  \(e_y\) the stationary generator, and let
  \(\varepsilon\in\{\pm1\}\) be fixed by the oriented handle-slide
  convention.  Then
  \begin{equation}
    \omega_x'=\omega_x+\varepsilon\omega_y,
    \qquad
    \omega_y'=\omega_y,
    \label{eq:two-two-marking-transport}
  \end{equation}
  while every uninvolved coordinate is unchanged.

  \item A \(1\)--\(2\) birth admits \(\omega'=\omega\oplus0\); every
  representative of the target class is cohomologous to one whose new
  \(2\)-handle coordinate is zero.

  \item For a \(2\)--\(3\) birth, the cocycle equation forces the new
  \(2\)-handle coordinate to be zero, so again
  \(\omega'=\omega\oplus0\).
\end{enumerate}
The death formulas are the inverses of the corresponding birth formulas.
\end{proposition}

\begin{proof}
Poincar\'e duality is natural, so the class attached to \(|m|_*\beta\) is
the cellular transport of the class attached to \(\beta\).  The standard
cellular map of each local move gives the formulas above: a \(2/2\) slide
sends the new moving generator to \(e_x+\varepsilon e_y\), while an
adjacent-index birth adds the corresponding acyclic two-term complex.  See
\cite[Sections~4.1--4.2]{DurstGeigesKegel2019}.
\end{proof}

\begin{theorem}[Marked Kirby--Cerf completeness]
\label{prop:marked-kirby-cerf}
Assume the following arrows, with the top edge deliberately absent:
\[
\begin{array}{ccc}
  |(\mathcal H,\beta)| && |(\mathcal H',\beta')| \\
  {\scriptstyle\rho}\!\downarrow &&
    \downarrow\!{\scriptstyle\rho'} \\
  (X,\alpha) & \xrightarrow{\ f\ } & (Y,\alpha').
\end{array}
\]
Here \(\rho\) and \(\rho'\) are marked Kirby presentations and \(f\) is a
marked diffeomorphism.  Then:
\begin{enumerate}[label=\textup{(\roman*)},ref=\textup{\roman*}]
  \item Ordinary Kirby--Cerf completeness, applied to the underlying
  unmarked arrows and without using \(\beta\) or \(\beta'\), produces a
  Kirby--Cerf movie
  \[
    M\colon
    \mathcal H=\mathcal H_0
    \xrightarrow{m_1}\mathcal H_1
    \longrightarrow\cdots\longrightarrow
    \mathcal H_N=\mathcal H'.
  \]
  Its realization \(\mu:=|M|\) supplies the missing top edge and completes the
  long diagram \eqref{eq:cerf-realizes-diffeomorphism} up to smooth isotopy.

  \item Writing \(\mu:=|M|\), the composite transport along the movie
  satisfies
  \[
    \mu_*\beta
    =\bigl(|m_N|_*\circ\cdots\circ|m_1|_*\bigr)(\beta)
    =\beta'.
  \]
  Namely, transporting the source marking through the independently
  constructed unmarked movie gives exactly the prescribed target marking,
  not merely another class represented on the same terminal Kirby datum.

  \item If both presentations are one-ended, \(M\) may be chosen
  one-ended throughout; equivalently, it uses the eight move types other
  than \(0\)--\(1\) and \(3\)--\(4\) birth/death.
\end{enumerate}
\end{theorem}

\begin{proof}
Ordinary completeness gives \(M\); functoriality of homology, together with
the markedness of \(\rho\), \(\rho'\), and \(f\), gives
\(\rho'_*\mu_*\beta=f_*\rho_*\beta=\alpha'=\rho'_*\beta'\), hence (ii)
because \(\rho'_*\) is an isomorphism; while
\cref{cor:one-ended-diffeomorphism-realization} gives (iii).
\end{proof}
