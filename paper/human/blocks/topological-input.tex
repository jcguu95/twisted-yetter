\subsection{From a marked four-manifold to marked handle data}
\label{subsec:topological-input}

\begin{definition}[Marked four-manifold]
\label{def:marked-four-manifold}
A \emph{marked four-manifold} is a pair \((X,\alpha)\), where \(X\) is a
closed, connected, oriented smooth four-manifold and
\(\alpha\in H_2(X;\mathbb Z)\).
\end{definition}

\begin{definition}[Kirby diagram]
\label{def:kirby-diagram}
A \emph{Kirby diagram} is a finite dotted--framed link
\[
  D_{\mathrm{Kir}}=D_{\mathrm{dot}}\sqcup D_{\mathrm{fr}}
  \subset S^3.
\]
The dotted sublink \(D_{\mathrm{dot}}\) is an unlink; its components encode
the \(1\)-handles attached to a single \(0\)-handle.  The integer-framed
components encode the attaching circles of the \(2\)-handles in the
resulting \(1\)-handlebody.  We write
\(W^{(2)}_{D_{\mathrm{Kir}}}\) for the represented \(0\)-, \(1\)-, and
\(2\)-handlebody.
\end{definition}

\begin{definition}[Special Kirby presentation]
\label{def:special-kirby-presentation}
Let \(X\) be a closed, connected, oriented smooth four-manifold.  A
\emph{special Kirby presentation} \(\mathcal H\) of \(X\) is a handle
decomposition with one \(0\)-handle and one \(4\)-handle, together with its
dotted--framed Kirby diagram \(D_{\mathcal H}\).  The diagram displays the
primal \(1\)- and \(2\)-handles; the \(3\)-handles close
\(W^{(2)}_{D_{\mathcal H}}\) before the final \(4\)-handle is attached.

After orienting the handles, the part of the integral handle complex relevant
to degree two is
\[
  C_3(\mathcal H)\xrightarrow{\partial_3}
  C_2(\mathcal H)\xrightarrow{\partial_2}
  C_1(\mathcal H).
  \tag{3.1}\label{eq:handle-complex}
\]
The oriented framed components give the preferred basis of
\(C_2(\mathcal H)\).  We write
\(h_3(\mathcal H)=\operatorname{rank}C_3(\mathcal H)\).
\end{definition}

\begin{convention}[Retained three-handle data]
\label{conv:retained-three-handle-data}
For a special Kirby presentation \(\mathcal H\), we retain from the
\(3\)-handles the integer \(h_3(\mathcal H)\) and the subgroup
\(\operatorname{im}(\partial_3)\subset C_2(\mathcal H)\).  We do not choose
explicit drawings of their attaching spheres.
\end{convention}

\begin{definition}[Handle representative of the marking]
\label{def:handle-homology-marking}
Let \((X,\alpha)\) be a marked four-manifold and let \(\mathcal H\) be a
special Kirby presentation of \(X\).  Set
\[
  C^2(\mathcal H):=\operatorname{Hom}(C_2(\mathcal H),\mathbb Z),
  \qquad
  Z^2(\mathcal H)
  :=\{a\in C^2(\mathcal H):a\circ\partial_3=0\},
\]
and
\[
  H^2(\mathcal H)
  :=Z^2(\mathcal H)/\operatorname{im}(\partial_2^*).
  \tag{3.2}\label{eq:handle-cohomology}
\]
Through the cellular comparison isomorphism of
\cref{prop:handle-homology-realization}, define \(\xi_{\mathcal H}\) to be
the class corresponding to \(\operatorname{PD}_X(\alpha)\).  A
\emph{handle cocycle for \(\alpha\)} is
an element \(a\in Z^2(\mathcal H)\) representing
\(\xi_{\mathcal H}\).

If \(e_k\) is the basis element represented by the \(k\)th framed component,
write
\[
  a_k:=a(e_k)\in\mathbb Z.
  \tag{3.3}\label{eq:handle-marking-coordinates}
\]
\end{definition}

\begin{proposition}[Existence and realization]
\label{prop:special-presentation-existence}
\label{prop:handle-homology-realization}
Let \(X\) be a closed, connected, oriented smooth four-manifold.  Then \(X\)
admits a special Kirby presentation.  For every such presentation
\(\mathcal H\) of \(X\), the handle complex
\eqref{eq:handle-complex} computes the integral (co)homology of \(X\); in
particular, cellular comparison gives a natural isomorphism
\[
  H^2(\mathcal H)\cong H^2(X;\mathbb Z).
\]
\end{proposition}

\begin{proof}
Choose a Morse function with a unique minimum and maximum and pass to its
handle decomposition.  Existence is
\cite[Proposition~4.2.13]{GompfStipsicz1999} and its dual argument.  A handle
decomposition is a relative CW decomposition, and its cellular chain complex
is the handle complex above; see the handle-homology discussion immediately
following that proposition.
\end{proof}
