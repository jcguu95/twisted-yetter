\subsubsection{Kirby calculus for closed oriented 4-manifolds}
\label{subsec:topological-input}

This subsection reviews textbook-level handle and Kirby calculus.  We follow
\cite[Chapters~4 and~5]{GompfStipsicz1999}; the original framed-link calculus
is due to Kirby \cite{Kirby1978}.  Throughout, 4-manifolds are closed,
connected, oriented, and smooth.  We do not impose a unique minimum or
maximum on a handle decomposition.

\begin{definition}[Handle attachment]
\label{def:handle-attachment}
Let \(W\) be an oriented smooth 4-manifold with boundary.  An
\emph{index-\(k\) handle attachment} is a smooth embedding
\[
  \phi\colon S^{k-1}\times D^{4-k}\hookrightarrow\partial W.
\]
It produces \(W\cup_{\phi}(D^k\times D^{4-k})\).  Several handles of the
same index are attached along a finite family of embeddings with pairwise
disjoint images.
\end{definition}

\begin{definition}[Kirby datum]
\label{def:kirby-datum}
A \emph{Kirby datum} \(\mathcal H\) consists of finite handle sets
\(I_k(\mathcal H)\), \(0\leq k\leq4\), and the following attaching data.
Begin with
\[
  W^{(0)}:=\bigsqcup_{q\in I_0(\mathcal H)}B_q^4.
\]
For \(k=1,2,3\), attach the handles successively along pairwise disjoint
embeddings
\[
  \phi_h^k\colon S^{k-1}\times D^{4-k}
  \hookrightarrow\partial W^{(k-1)},
  \qquad h\in I_k(\mathcal H),
\]
and denote the resulting manifold by \(W^{(k)}\).  Finally, for every
\(q\in I_4(\mathcal H)\), choose an orientation-reversing diffeomorphism
\[
  \gamma_q\colon \partial B_q^4\xrightarrow{\ \cong\ }Y_q,
\]
where the \(Y_q\) are precisely the connected components of
\(\partial W^{(3)}\).  Thus the closing maps exhaust the boundary, with one
map for each boundary component.  The resulting closed manifold is required
to be connected.
\end{definition}

\begin{definition}[Realization and presentation]
\label{def:kirby-presentation}
The \emph{realization} of \(\mathcal H\) is
\[
  |\mathcal H|:=W^{(3)}\cup_{\bigsqcup_q\gamma_q}
    \bigsqcup_{q\in I_4(\mathcal H)}B_q^4.
\]
A \emph{Kirby presentation} of a closed connected oriented smooth
4-manifold \(X\) is a pair
\[
  (\mathcal H,\rho),\qquad
  \rho\colon|\mathcal H|\xrightarrow{\ \cong\ }X,
\]
with \(\rho\) orientation preserving.  The choice of any individual closing
map does not affect the oriented diffeomorphism type: two such maps differ by
an orientation-preserving diffeomorphism of \(S^3\), which extends over
\(B^4\) by Cerf's theorem \(\Gamma_4=0\) \cite{Cerf1968}.
\end{definition}

\begin{definition}[Derived handle chain and cochain complexes]
\label{def:derived-handle-chain-complex}
Put \(W^{(-1)}=\varnothing\) and \(W^{(4)}=|\mathcal H|\), and define
\[
  C_k(\mathcal H):=
  H_k\bigl(W^{(k)},W^{(k-1)};\mathbb Z\bigr).
\]
After orienting the handle cores, their classes identify
\[
  C_k(\mathcal H)\cong
  \bigoplus_{h\in I_k(\mathcal H)}\mathbb Z[h].
\]
The boundary maps are the connecting maps of the handle filtration; their
matrix entries are the signed intersections of attaching and belt spheres.
In particular,
\begin{equation}
  C_3(\mathcal H)\xrightarrow{\partial_3}
  C_2(\mathcal H)\xrightarrow{\partial_2}
  C_1(\mathcal H).
  \label{eq:handle-complex}
\end{equation}
Write \(h_k(\mathcal H)=|I_k(\mathcal H)|\).  The chain complex and the
numbers \(h_k\) are derived from the handle attachments; they are not extra
parts of a Kirby datum.
\end{definition}

\begin{definition}[One-ended Kirby datum]
\label{def:one-ended-kirby-datum}
A Kirby datum is \emph{one-ended} if
\(h_0(\mathcal H)=h_4(\mathcal H)=1\).  Its \(0\)-handle is identified with
the standard \(B^4\), whose boundary is the standard \(S^3\).  The \(1\)- and
\(2\)-handle attaching data then admit the usual dotted--framed record
\[
  D_{\mathcal H}=D_{\mathrm{dot}}\sqcup D_{\mathrm{fr}}\subset S^3.
\]
Here \(D_{\mathrm{dot}}\) is a finite dotted unlink.  Each of its components
is an unknot encoding one pair of disjoint attaching \(3\)-balls for a
\(1\)-handle, in the standard dotted-circle notation.  Surgery on all dotted
components identifies the resulting three-manifold with the boundary of the
\(0\)--\(1\) subhandlebody.

The set \(D_{\mathrm{fr}}\) is a framed link in that boundary, displayed in
the original \(S^3\) together with \(D_{\mathrm{dot}}\).  Its components are
the attaching circles of the \(2\)-handles.  The framing of a component is
the homotopy class of a trivialization of its normal \(2\)-plane bundle in
the boundary; an integer records this class after the usual diagrammatic
reference framing has been chosen.  Thus a strand of
\(D_{\mathrm{fr}}\) may pass through a spanning disk of a dotted component,
but the two links themselves are disjoint.  The \(3\)-handle attaching data
are not encoded by this dotted--framed link.  Consequently
\(D_{\mathcal H}\) is a reduced record used for calculations, not the
definition of the full Kirby datum.

Neither \(D_{\mathrm{dot}}\) nor \(D_{\mathrm{fr}}\) is intrinsically
oriented.  Orientations chosen later for their components are auxiliary:
they select bases for the handle chain groups and oriented representatives
for categorical labels, but are not additional Kirby data.
\end{definition}

\begin{proposition}[Existence and one-ended reduction]
\label{prop:kirby-presentation-existence}
Every closed connected oriented smooth 4-manifold admits a Kirby
presentation.  Moreover, every Kirby presentation can be changed, by
same-index slides and cancellations of \(0\)--\(1\) and \(3\)--\(4\) pairs,
to a one-ended Kirby presentation of the same manifold.
\end{proposition}

\begin{proof}
A Morse function gives the first assertion.  Since the realized manifold is
connected, the incidence graph of its \(0\)- and \(1\)-handles is connected.
Choose a spanning tree, slide the corresponding \(1\)-handles into cancelling
position, and cancel all but one \(0\)-handle.  Apply the same argument to the
turned-over decomposition to cancel all but one \(4\)-handle.  This is the
standard reduction to a handle decomposition with one minimum and one maximum;
see \cite[Sections~4.2 and~5.1]{GompfStipsicz1999}.
\end{proof}

\begin{definition}[Handle slide]
\label{def:handle-slide}
Let the handles of index less than \(k\) already be attached.
A \emph{same-index \(k/k\) slide} replaces the attaching sphere of a moving
index-\(k\) handle by its boundary connected sum with a framed parallel copy
of the attaching sphere of a stationary index-\(k\) handle.  The connected
sum is taken along an embedded band in the boundary of the lower
subhandlebody, disjoint from all other attaching regions.  The normal
framing is the band sum of the two given framings.  All other attaching data
are unchanged outside a neighborhood of this band.

\end{definition}

\begin{remark}[Level isotopy in dotted notation]
An ambient isotopy of a \(2\)-handle attaching circle in the boundary of the
\(0\)--\(1\) subhandlebody may be drawn in the \(S^3\) dotted-circle picture
by passing a strand through a dotted component.  This is a depiction of level
isotopy, not an additional Kirby--Cerf move and not a slide of an index-\(2\)
handle over an index-\(1\) handle.
\end{remark}

\begin{definition}[Cancelling handle pair]
\label{def:cancelling-handle-pair}
An adjacent pair consisting of an index-\(k\) handle \(h\) and an
index-\((k+1)\) handle \(H\) is \emph{cancelling} if the attaching sphere of
\(H\) meets the belt sphere of \(h\) transversely in exactly one point and
the attaching framing of \(H\) agrees there with the product framing of the
standard cancelling local model.  A \emph{death} removes such an isolated
standard pair; a \emph{birth} is the inverse operation.  Handles that meet
the local model must first be moved away by isotopy or slides.

For a \(1\)--\(2\) pair in dotted notation, this says that the \(2\)-handle
attaching circle goes geometrically once over the dotted \(1\)-handle and
has the standard relative framing.  The definitions for \(0\)--\(1\),
\(2\)--\(3\), and \(3\)--\(4\) pairs are the corresponding attaching-sphere
and belt-sphere conditions.
\end{definition}

\begin{definition}[Elementary Kirby--Cerf move]
\label{def:kirby-move}
An elementary Kirby--Cerf move \(m\colon\mathcal H\to\mathcal H'\) is a
level isotopy, a same-index handle slide as in \cref{def:handle-slide}, the
birth or death of a pair satisfying \cref{def:cancelling-handle-pair}.  In
dimension four these are the following twelve directed types:
\begin{enumerate}[label=(\roman*),ref=\roman*]
  \item\label{move:framed-isotopy} level isotopy of attaching data;
  \item\label{move:one-over-one} a \(1/1\) slide;
  \item\label{move:two-over-two} a \(2/2\) slide;
  \item\label{move:three-over-three} a \(3/3\) slide;
  \item a \(0\)--\(1\) birth;
  \item a \(0\)--\(1\) death;
  \item\label{move:one-two-birth} a \(1\)--\(2\) birth;
  \item\label{move:one-two-death} a \(1\)--\(2\) death;
  \item\label{move:two-three-birth} a \(2\)--\(3\) birth;
  \item\label{move:two-three-death} a \(2\)--\(3\) death;
  \item a \(3\)--\(4\) birth; or
  \item a \(3\)--\(4\) death.
\end{enumerate}
There is no \(3/2\) slide in this list.  A handle slide is intrinsically a
same-index \(k/k\) operation.  Isotopies which look like a strand passing
through a dotted circle are already included under level isotopy.
\end{definition}

\begin{proposition}[Realization of an elementary move]
\label{prop:elementary-move-realization}
Every elementary move \(m\colon\mathcal H\to\mathcal H'\) in
\cref{def:kirby-move} determines, up to isotopy relative to the complement
of its supporting neighborhood, an orientation-preserving diffeomorphism
\[
  |m|\colon |\mathcal H|\longrightarrow|\mathcal H'|.
\]
For a same-index slide this is the standard handle-slide diffeomorphism; for
a birth or death it is the diffeomorphism supplied by cancellation of the
standard local pair; and for a level isotopy it is the induced ambient
isotopy of the corresponding attaching data.
\end{proposition}

\begin{proof}
These are the local realization statements of handle calculus.  The
attaching-sphere and framing descriptions above are exactly the hypotheses
of the handle-slide and handle-cancellation theorems.  The maps are the
identity outside the displayed supporting neighborhoods.  In one-ended
dotted notation the corresponding handle-slide pictures are the
dotted-circle slide and the oriented framed band sum of
\cite[Figures~5.39 and~5.36 and Section~5.1]{GompfStipsicz1999}.
A strand drawn as passing through a dotted circle instead represents an
ambient isotopy of the attaching data.
\end{proof}

\paragraph{Kirby--Cerf movies.}
A \emph{Kirby--Cerf movie} \(M\) from \(\mathcal H_0\) to
\(\mathcal H_N\) is a
finite composable sequence of elementary moves
\[
  \mathcal H_0\xrightarrow{m_1}\mathcal H_1
  \longrightarrow\cdots\longrightarrow
  \mathcal H_{N-1}\xrightarrow{m_N}\mathcal H_N.
\]
Its realization is the orientation-preserving diffeomorphism
\[
  |M|:=|m_N|\circ\cdots\circ|m_1|
  \colon|\mathcal H_0|\longrightarrow|\mathcal H_N|.
\]
The empty movie has realization \(\operatorname{id}_{|\mathcal H_0|}\).

\begin{theorem}[Kirby--Cerf completeness]
\label{prop:handle-movie-completeness}
Let \((\mathcal H_0,\rho_0)\) and \((\mathcal H_1,\rho_1)\) be Kirby
presentations of \(X_0\) and \(X_1\), and let
\(f\colon X_0\to X_1\) be an orientation-preserving diffeomorphism.  There
is a Kirby--Cerf movie
\[
  M\colon\mathcal H_0=\mathcal K_0
  \xrightarrow{m_1}\mathcal K_1
  \longrightarrow\cdots\longrightarrow
  \mathcal K_N=\mathcal H_1
\]
and choices of the realization maps in
\cref{prop:elementary-move-realization} such that, for
\(\mu:=|M|\),
the diagram
\begin{equation*}
\label{eq:cerf-realizes-diffeomorphism}
\begin{array}{ccccccc}
  |\mathcal H_0| & \xrightarrow{|m_1|} & |\mathcal K_1|
    & \xrightarrow{\ \cdots\ } & |\mathcal K_{N-1}|
    & \xrightarrow{|m_N|} & |\mathcal H_1| \\
  {\scriptstyle\rho_0}\!\downarrow &&&&&&
    \downarrow\!{\scriptstyle\rho_1} \\
  X_0 & \multicolumn{5}{c}{\xrightarrow{\hspace{9em} f\hspace{9em}}} & X_1
\end{array}
\qquad\text{commutes up to smooth isotopy.}
\tag{KC}
\end{equation*}
Precisely, there is an isotopy, that is, a smooth one-parameter family
\(F_t\colon|\mathcal H_0|\to X_1\), \(0\leq t\leq1\), of
orientation-preserving diffeomorphisms such that
\[
  F_0=\rho_1\circ\mu,
  \qquad
  F_1=f\circ\rho_0.
\]
The movie itself is a finite discrete sequence.  The parameter \(t\) belongs
to the isotopy comparing the composite realization \(|M|\) with the
prescribed diffeomorphism \(f\); it does not index additional Kirby moves.
Thus the Kirby movie realizes the prescribed isotopy class of \(f\).
\end{theorem}

\begin{proof}
Choose Morse functions with pseudo-gradients realizing \(\mathcal H_0\) and
\(\mathcal H_1\).  Transfer the latter Morse data to
\(|\mathcal H_0|\) by the diffeomorphism
\[
  \varphi:=\rho_1^{-1}\circ f\circ\rho_0
  \colon|\mathcal H_0|\longrightarrow|\mathcal H_1|.
\]
A generic path from the first Morse data to the transferred second Morse
data meets only the codimension-one strata of Cerf theory.  The regular
intervals give level isotopies, exchanges of critical values reorder
disjoint handle attachments, isolated same-index connecting trajectories in
the pseudo-gradient family give handle slides, and the birth--death strata
  give cancelling adjacent-index pairs.  In dotted notation a level isotopy
  may be drawn as a sequence of strand motions through dotted circles, but
  these are not additional Kirby--Cerf moves.  This produces the finite movie
  above.

Track the level identifications and pseudo-gradient flows throughout the
path.  Their composite identifies the terminal handle decomposition with
the one transferred by \(\varphi\), and hence is isotopic to \(\varphi\).
After composing with \(\rho_1\), this is precisely
\eqref{eq:cerf-realizes-diffeomorphism}.  This is the realization-aware form
of the standard Cerf comparison theorem; see \cite{Cerf1970} and
\cite[Theorem~4.2.12]{GompfStipsicz1999}.
\end{proof}

\begin{corollary}[One-ended realization]
\label{cor:one-ended-diffeomorphism-realization}
If both presentations in \cref{prop:handle-movie-completeness} are
one-ended, the realizing movie may be chosen one-ended throughout.  Hence it
uses the eight move types other than \(0\)--\(1\) and \(3\)--\(4\)
birth/death.
\end{corollary}

\begin{proof}
This is the connected Cerf refinement of
\cite[Remark~2.7 and Theorem~3.4]{GayWehrheimWoodward2012}, applied to the
generic path used in the proof of
\cref{prop:handle-movie-completeness}.  Tracking the same level
identifications preserves
\eqref{eq:cerf-realizes-diffeomorphism}.
\end{proof}
