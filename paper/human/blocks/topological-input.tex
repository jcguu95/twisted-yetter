\subsection{Kirby calculus for closed oriented four-manifolds}
\label{subsec:topological-input}

This subsection is a review of textbook-level handle and Kirby calculus.  We
follow \cite[Chapters~4 and~5]{GompfStipsicz1999}; the original framed-link
calculus is due to Kirby \cite{Kirby1978}.  Throughout, the four-manifolds are
closed, connected, oriented, and smooth.

\begin{definition}[Handle attachment]
\label{def:handle-attachment}
Let \(W\) be an oriented smooth four-manifold with boundary.  An
\emph{index-\(k\) handle attachment} is a smooth embedding
\[
  \phi\colon S^{k-1}\times D^{4-k}\hookrightarrow\partial W.
\]
It produces the oriented manifold
\[
  W\cup_{\phi}(D^k\times D^{4-k}),
\]
where \(S^{k-1}\times D^{4-k}\) is the attaching region of the handle.
Several handles of the same index are attached along a family of embeddings
with pairwise disjoint images.
\end{definition}

\begin{definition}[Special Kirby datum]
\label{def:special-kirby-datum}
A \emph{special Kirby datum} \(\mathcal H\) consists of three finite labelled
families of attaching embeddings, constructed successively as follows.
Write \(\Phi^k\) for the disjoint union of all index-\(k\) attaching
embeddings.  Start with \(W^{(0)}=B^4\), and set
\begin{align*}
  \phi_i^1&\colon S^0\times D^3\hookrightarrow
    \partial W^{(0)}=S^3,
  &W^{(1)}&:=W^{(0)}\cup_{\Phi^1}
    \bigsqcup_i(D^1\times D^3),\\
  \phi_j^2&\colon S^1\times D^2\hookrightarrow
    \partial W^{(1)},
  &W^{(2)}&:=W^{(1)}\cup_{\Phi^2}
    \bigsqcup_j(D^2\times D^2),\\
  \phi_\ell^3&\colon S^2\times D^1\hookrightarrow
    \partial W^{(2)},
  &W^{(3)}&:=W^{(2)}\cup_{\Phi^3}
    \bigsqcup_\ell(D^3\times D^1).
\end{align*}
We require \(\partial W^{(3)}\) to be orientation-preservingly
diffeomorphic to \(S^3\).  The unique \(0\)-handle and the final
\(4\)-handle are therefore implicit rather than additional entries of the
datum.

After orienting the handles, the part of the integral handle complex relevant
to degree two is
\[
  C_3(\mathcal H)\xrightarrow{\partial_3}
  C_2(\mathcal H)\xrightarrow{\partial_2}
  C_1(\mathcal H).
  \tag{3.1}\label{eq:handle-complex}
\]
The oriented framed components give the preferred basis of
\(C_2(\mathcal H)\).  We write
\(h_3(\mathcal H)=\operatorname{rank}C_3(\mathcal H)\).
\end{definition}

\begin{definition}[Realization and presentation]
\label{def:special-kirby-presentation}
Choose an orientation-reversing diffeomorphism
\(\partial B^4\to\partial W^{(3)}\) and define the \emph{realization}
\[
  |\mathcal H|:=W^{(3)}\cup_{S^3}B^4.
\]
Its oriented diffeomorphism type is independent of the closing map: two
closing maps differ by an orientation-preserving diffeomorphism of \(S^3\),
which extends over \(B^4\) by Cerf's theorem \(\Gamma_4=0\)
\cite{Cerf1968}.  A
\emph{special Kirby presentation} of a closed, connected, oriented smooth
four-manifold \(X\) is a pair
\[
  (\mathcal H,\rho),
  \qquad
  \rho\colon|\mathcal H|\xrightarrow{\ \cong\ }X
\]
with \(\rho\) orientation preserving.  When the realization map is
understood, we suppress \(\rho\) and simply call \(\mathcal H\) a
presentation of \(X\).
\end{definition}

\begin{remark}[Reduced records]
\label{rem:reduced-kirby-records}
The preceding definitions retain the full attaching embeddings.  Up to
attaching isotopy, less information is usually drawn.  For each labelled
\(1\)-handle, \(S^0\times D^3\) is represented by the two centres of its
attaching balls together with their pairing; the ball neighbourhoods are
standard.  A \(2\)-handle is represented by its attaching circle and integer
framing.  A \(3\)-handle is represented by its attaching sphere with its
product neighbourhood.  The first two reduced records form the usual finite
dotted--framed link
\[
  D_{\mathcal H}=D_{\mathrm{dot}}\sqcup D_{\mathrm{fr}}\subset S^3.
\]
These reductions are notation for the full datum, not replacements for its
definition.
\end{remark}

\begin{proposition}[Existence of special presentations]
\label{prop:special-presentation-existence}
Let \(X\) be a closed, connected, oriented smooth four-manifold.  Then \(X\)
admits a special Kirby presentation.
\end{proposition}

\begin{proof}
Choose a Morse function with a unique minimum and maximum and pass to its
handle decomposition.  This is \cite[Proposition~4.2.13]{GompfStipsicz1999}
together with its dual argument.
\end{proof}

\begin{definition}[Elementary Kirby--Cerf move]
\label{def:kirby-move}
Let \(\mathcal H\) and \(\mathcal H'\) be special Kirby data.  An
elementary Kirby--Cerf move
\(m\colon\mathcal H\to\mathcal H'\) is one of the following nine directed
modifications of the complete handle data:
\begin{enumerate}[label=(\roman*),ref=\roman*]
  \item\label{move:framed-isotopy} a level isotopy of attaching data, seen
  as framed isotopy when it changes the dotted--framed diagram;
  \item\label{move:one-over-one} a \(1/1\) slide;
  \item\label{move:two-over-one} a \(2/1\) slide;
  \item\label{move:two-over-two} a \(2/2\) slide;
  \item\label{move:three-over-three} a \(3/3\) slide;
  \item\label{move:one-two-birth} a \(1\)--\(2\) birth;
  \item\label{move:one-two-death} a \(1\)--\(2\) death;
  \item\label{move:two-three-birth} a \(2\)--\(3\) birth; or
  \item\label{move:two-three-death} a \(2\)--\(3\) death.
\end{enumerate}
For an \(r/s\) slide, \(r\) is the index of the moving handle and \(s\) the
index of the stationary handle.  Outside the supporting ball or
cancelling-pair neighborhood, the handle decomposition is unchanged.
Level isotopy and a \(3/3\) slide act on the actual \(3\)-handle attaching
embeddings even though that change is invisible in the dotted--framed
diagram.
\end{definition}

\begin{convention}[Standard local models]
\label{conv:referenced-kirby-dictionary}
The names in \cref{def:kirby-move} have their standard meanings in Kirby
calculus.  The \(1/1\) and \(2/1\) moves are the dotted-circle moves in
\cite[Figures~5.39 and~5.36]{GompfStipsicz1999}; a \(2/2\) move is an oriented
band sum with a framed push-off of the stationary component
\cite[Section~5.1]{GompfStipsicz1999}; a \(3/3\) move is the corresponding
same-index slide of one \(3\)-handle attaching sphere over another; and
births insert the isolated standard cancelling pairs, with deaths as their
inverses.  This convention fixes the topology of each source and target and
nothing more.
\end{convention}

\begin{theorem}[Kirby--Cerf completeness for special presentations]
\label{prop:special-handle-movie-completeness}
Let \((\mathcal H_0,\rho_0)\) and \((\mathcal H_1,\rho_1)\) be special Kirby
presentations of closed, connected, oriented smooth four-manifolds \(X_0\)
and \(X_1\), and let
\(f\colon X_0\to X_1\) be an orientation-preserving diffeomorphism.  Via
\(\rho_0\) and \(f^{-1}\circ\rho_1\), regard both handle decompositions as
embedded decompositions of the fixed manifold \(X_0\).  They are related by
a finite handle movie on \(X_0\),
\[
  \mathcal H_0\rightsquigarrow f^*\mathcal H_1,
\]
consisting of the nine moves in \cref{def:kirby-move}.  Here
\(f^*\mathcal H_1\) denotes the second decomposition viewed inside \(X_0\),
not a claim that the movie alone realizes the mapping class of \(f\).
The prescribed map appears as the final tautological endpoint identification
\[
  f\colon f^*\mathcal H_1\longrightarrow\mathcal H_1
\]
between the two ambient manifolds.
\end{theorem}

\begin{proof}
Pull \(\mathcal H_1\) back along \(f\), so that it becomes a handle
decomposition of \(X_0\).  Choose small balls inside the unique \(0\)- and
\(4\)-handles and identify their collars.  Removing the interiors of these
balls turns both decompositions into relative handle decompositions of the
same cobordism \(S^3\to S^3\), using only handles of indices \(1,2,3\).

We use connected Cerf theory only to keep all intermediate levels connected:
\cite[Remark~2.7 and Theorem~3.4]{GayWehrheimWoodward2012} supplies a connected Cerf
decomposition and hence excludes \(0\)--\(1\) and \(3\)--\(4\) births.  That
result deliberately does not record attaching cycles or handle slides.  For
those data we use Cerf's original one-parameter theory \cite{Cerf1970} in the
standard relative handle-completeness form
\cite[Theorem~4.2.12]{GompfStipsicz1999}: two relative handle decompositions
are related by level isotopies of attaching data, same-index handle slides,
and births or deaths of adjacent-index cancelling pairs.

In dotted--framed notation, index-\(1\) slides are the \(1/1\) move
\cite[Figure~5.39]{GompfStipsicz1999}, isotopies through the \(1\)-handlebody
are generated by framed isotopy and the \(2/1\) move
\cite[Figure~5.36]{GompfStipsicz1999}, and index-\(2\) slides are the framed
band sums of \cite[Section~5.1]{GompfStipsicz1999}.  The two possible
adjacent-index cancellations give the four birth/death moves.  Thus the Cerf
movie has precisely the asserted form: same-index slides include the
\(3/3\) move on the full \(3\)-handle attaching data, and level isotopies
include isotopies of those attaching spheres.  Every operation takes place
on the fixed manifold \(X_0\).  This proves the asserted object-level
completeness.  The final map \(f\) is supplied by the chosen realization of
the endpoint; no mapping-class realization theorem for the move sequence is
being asserted.
\end{proof}

\begin{remark}[Mapping-class refinement]
\label{rem:mapping-class-refinement}
The preceding theorem is deliberately an object-level completeness statement.
It does not assert that a based Kirby--Cerf movie realizes an arbitrarily
prescribed element of \(\pi_0\operatorname{Diff}^+(X)\).  To the author's
knowledge, it remains open in this generality whether every such mapping class
admits a based-movie realization and whether isotopies between the resulting
diffeomorphisms are generated by a complete list of two-parameter
``moves of moves.''  Related partial results for \(S^4\), formulated using
loops of embedded spheres and Cerf theory, appear in
\cite{Gay2025}.  No such mapping-class refinement is used below.
\end{remark}
