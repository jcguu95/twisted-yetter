\subsection{The presentation-level evaluator}
\label{subsec:graphical-construction}

\begin{definition}[Homologically decorated handle morphism]
\label{def:decorated-kirby-graph}
\label{def:dotted-circle-translation}
Assume the following data.
\begin{enumerate}[label=\textup{(\roman*)}]
\item a premodular fusion category \(\cC\) over \(\mathbb C\);
\item a monoidal natural automorphism
  \(\eta\colon\operatorname{Id}_{\cC}\Rightarrow
  \operatorname{Id}_{\cC}\);
\item a one-ended Kirby presentation \(\mathcal H\), whose dotted--framed
  link lies in the standard \(S^3=\partial B^4\);
\item a handle cocycle \(\omega\in Z^2(\mathcal H)\);
\item auxiliary orientations, an admissible dotted-handle section system,
  composition-dual bases for the tensor-unit projectors, and a regular planar
  drawing; denote this collection by \(\mathfrak a\).
\end{enumerate}
Write \(F_2(\mathcal H)\) for the finite set of framed components, indexed
as \(K_k\), and put
\[
  \operatorname{Lab}_{\cC}(\mathcal H)
  :=\prod_{k\in F_2(\mathcal H)}\operatorname{Irr}(\cC).
\]
Thus \(\lambda\in\operatorname{Lab}_{\cC}(\mathcal H)\) assigns a simple
object \(\lambda_k\) to each framed component.  For a closed
\(\cC\)-coloured ribbon graph \(G\subset S^3\), write
\[
  \operatorname{ev}_{\cC}(G)
  \in\operatorname{End}_{\cC}(\mathbf1)
\]
for its graphical-interpretation morphism from
\cref{def:rt-graph-evaluation}.

If \(e_k\) is the oriented \(2\)-handle generator selected by
\(\mathfrak a\), put \(\omega_k:=\omega(e_k)\).  Colour \(K_k\) by
\(\lambda_k\) and place on it the coupon
\[
  (\eta^{\omega_k})_{\lambda_k}
  =\eta_{\lambda_k}^{\omega_k}\operatorname{id}_{\lambda_k},
\]
where \(\eta_{\lambda_k}\in\mathbb C^\times\) is the scalar of
\cref{lem:simple-object-character-scalars}.  A dotted component \(\delta\)
is not coloured: cut its handle, insert
\[
  P_{\mathbf1,B_\delta}\colon B_\delta\longrightarrow B_\delta
\]
on the ordered boundary object, and reglue.  The resulting closed ribbon
diagram is denoted by
\(G^\eta_{\lambda,\omega}(\mathcal H;\mathfrak a)\), and
\[
  L^\eta_{\lambda,\omega}(\mathcal H;\mathfrak a)
  :=\operatorname{ev}_{\cC}\bigl(
       G^\eta_{\lambda,\omega}(\mathcal H;\mathfrak a)\bigr)
  \in\operatorname{End}_{\cC}(\mathbf1).
\]
The Kirby-colour coefficient
\(\prod_{k\in F_2(\mathcal H)}d_{\lambda_k}\) is not included in this
notation.
\end{definition}

\begin{proposition}[Auxiliary-choice independence]
\label{prop:graphical-auxiliary-independence}
The endomorphism family
\(L^\eta_{\lambda,\omega}(\mathcal H;\mathfrak a)\) is independent of the
projector bases, section system, point at infinity, and planar projection.
Under reversal of a framed-component orientation, the precise reindexing is
\[
  (e_k,\omega_k,\lambda_k)
  \longmapsto(-e_k,-\omega_k,\lambda_k^*).
\]
\end{proposition}

\begin{proof}
The underlying ribbon identities are the fixed-presentation checks in
\cite[Definition~5.11 and Lemma~2.28]{Tham2021}; the only extra morphisms are
the character coupons.  We check every auxiliary choice explicitly.
\begin{itemize}
\item \textbf{Projector bases.}
Changing a composition-dual basis does not move or alter any character
coupon.  Both decorated expressions use the same canonical endomorphism
\(P_{\mathbf1,B_\delta}\), by
\cref{lem:tensor-unit-projector-basis-independence}.

\item \textbf{Dotted-handle sections.}
Choose a generic isotopy between the two section systems.  Its trace meets
the compact decorated graph in finitely many elementary ribbon events.  Move
the finitely many coupons away from each next event by naturality.  Across
that event, transport the projector by
\cref{lem:unit-projector-naturality} and the coupon by
\cref{lem:finite-coupon-transport}.  Induction over the event list gives the
same reglued morphism.  This does not assert that a projector absorbs
arbitrary coupons with unrelated exponents.

\item \textbf{Point at infinity and planar projection.}
The embedded decorated ribbon graph in \(S^3\) is unchanged.  Since every
character coupon is a morphism of \(\cC\), coloured-ribbon-graph isotopy
invariance applies without modification; see
\cite[Propositions~3.33--3.34]{Tham2021} or
\cite[Theorem~6.1]{Shum1994}.

\item \textbf{Orientation of a framed component.}
Reversing \(K_k\) sends \((e_k,\omega_k,\lambda_k)\) to
\((-e_k,-\omega_k,\lambda_k^*)\).  The new coupon is
\((\eta^{-\omega_k})_{\lambda_k^*}\).  By
\cref{lem:character-identities},
\(\eta_{\lambda_k^*}=\eta_{\lambda_k}^{-1}\), so this is exactly the
pivotal dual of \((\eta^{\omega_k})_{\lambda_k}\).

\item \textbf{Orientation of a dotted component.}
No character exponent is assigned to a dotted component.  Reversing its
orientation replaces its boundary word and tensor-unit projector by their
pivotal duals; the framed-ribbon coupons are carried along by the same
canonical duality identification.
\end{itemize}
\end{proof}

\begin{definition}[Intrinsic decorated Kirby morphism]
\label{def:intrinsic-decorated-kirby-morphism}
By \cref{prop:graphical-auxiliary-independence}, the common value
\[
  L^\eta_{\lambda,\omega}(\mathcal H)
  \in\operatorname{End}_{\cC}(\mathbf1)
\]
is well defined without \(\mathfrak a\).  Under
\(\operatorname{End}_{\cC}(\mathbf1)\cong\mathbb C\), denote its scalar by
\(\langle L^\eta_{\lambda,\omega}(\mathcal H)\rangle\).
\end{definition}

\begin{definition}[Homologically twisted Crane--Yetter evaluator]
\label{def:twisted-handle-evaluator}
Assume the following data.
\begin{enumerate}[label=\textup{(\roman*)}]
\item a premodular fusion category \(\cC\) over \(\mathbb C\);
\item a monoidal natural automorphism
  \(\eta\colon\operatorname{Id}_{\cC}\Rightarrow
  \operatorname{Id}_{\cC}\);
\item a one-ended Kirby presentation \(\mathcal H\);
\item a handle cocycle \(\omega\in Z^2(\mathcal H)\).
\end{enumerate}
Write \(F_2(\mathcal H)\) for the finite set of framed components and
\[
  \operatorname{Lab}_{\cC}(\mathcal H)
  =\prod_{k\in F_2(\mathcal H)}\operatorname{Irr}(\cC)
\]
for its simple-label states, as in \cref{def:decorated-kirby-graph}.
The resulting scalar is
\begin{equation}
  \boxed{
  \CY_{\cC,\eta}(\mathcal H,\omega)
  :=
  \operatorname{Dim}(\cC)^{1-h_3(\mathcal H)}
  \sum_{\lambda\in\operatorname{Lab}_{\cC}(\mathcal H)}
    \left(\prod_{k\in F_2(\mathcal H)} d_{\lambda_k}\right)
      \bigl\langle L^\eta_{\lambda,\omega}(\mathcal H)\bigr\rangle.}
  \label{eq:twisted-handle-evaluator}
\end{equation}
In this formula:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \(\lambda\) ranges over
  \(\operatorname{Lab}_{\cC}(\mathcal H)\), the assignments of simple
  objects to the framed components;
\item \(d_{\lambda_k}\) is the quantum dimension of the label on \(K_k\);
\item \(\operatorname{Dim}(\cC)=\sum_i d_i^2\) is the global dimension;
\item \(h_3(\mathcal H)\) is the number of \(3\)-handles;
\item \(\langle-\rangle\) is the scalar identification of
  \cref{def:intrinsic-decorated-kirby-morphism}.
\end{enumerate}
\end{definition}

\begin{remark}[Identity-character specialization]
\label{rem:identity-character-specialization}
If \(\eta=\operatorname{id}\), every character coupon is an identity and
\(\CY_{\cC,\operatorname{id}}(\mathcal H,\omega)\) is independent of
\(\omega\).  It is the closed one-ended handle formula obtained from
\cite[Definition~5.11]{Tham2021} on the reversed handle decomposition.
In that formula, the unique index-\(0\) handle contributes
\(\operatorname{Dim}(\cC)\), each index-\(1\) handle contributes
\(\operatorname{Dim}(\cC)^{-1}\), the index-\(2\) handles contribute the
regular-colour sums, and the index-\(3\) handles contribute the tensor-unit
projections.  Reversing the primal decomposition sends its unique
\(4\)-handle to index \(0\) and its \(h_3(\mathcal H)\) many \(3\)-handles
to index \(1\).  Hence the scalar factors outside the graph multiply to
\[
  \operatorname{Dim}(\cC)\,
  \operatorname{Dim}(\cC)^{-h_3(\mathcal H)}
  =\operatorname{Dim}(\cC)^{1-h_3(\mathcal H)}.
\]
In this specialization,
Kirby--Cerf invariance is \cite[Proposition~5.13]{Tham2021}; agreement with
the PLCW and original Crane--Yetter state sums is
\cite[Theorems~5.26 and~3.61]{Tham2021}.
\end{remark}

\begin{proposition}[Dependence on the three-handle data]
\label{prop:evaluator-three-handle-dependence}
Although a one-ended Kirby presentation contains the full attaching embeddings
of its \(3\)-handles, the marked evaluator uses that data only through
\[
  h_3(\mathcal H)
  \quad\text{and}\quad
  \operatorname{im}(\partial_3)\subset C_2(\mathcal H).
\]
More precisely, the scalar in
\eqref{eq:twisted-handle-evaluator} reads \(h_3(\mathcal H)\) in its
normalization, while the requirement that
\(\omega\in Z^2(\mathcal H)\) depends on \(\partial_3\) only through the
condition that \(\omega\) vanish on \(\operatorname{im}(\partial_3)\).
\end{proposition}

\begin{proof}
The coloured ribbon graph and its label sum are constructed entirely from
the dotted--framed \(1\)- and \(2\)-handle diagram.  The displayed formula
contains the \(3\)-handles explicitly only in the exponent
\(1-h_3(\mathcal H)\).  Finally,
\[
  \omega\in Z^2(\mathcal H)
  \quad\Longleftrightarrow\quad
  \omega\circ\partial_3=0,
\]
which depends only on the image of \(\partial_3\).  No other part of the
attaching embeddings enters the formula.
\end{proof}

\begin{proposition}[Finiteness of the evaluator]
\label{prop:evaluator-finiteness}
For a one-ended Kirby datum, the sum in
\eqref{eq:twisted-handle-evaluator} is finite.
\end{proposition}

\begin{proof}
The Kirby-colour sums are finite because \(\cC\) has finitely many simple
isomorphism classes.  Each tensor-unit projector is a finite sum because
\(\cC\) is finite semisimple.  A Kirby diagram has finitely many components,
so only finitely many choices occur.
\end{proof}

\begin{remark}[What the dotted circle is not]
\label{rem:dotted-circle-operation-scope}
A dotted circle is not a coloured ribbon component and is never assigned the
Kirby colour.  It denotes the projector onto the tensor-unit through-channel.
In a modular category that projector admits an alternative encircling
description using \(\operatorname{Dim}(\cC)^{-1}\Omega_{\cC}\), but no such
replacement is used in the premodular construction.
\end{remark}

\begin{remark}[Intrinsic meaning]
\label{rem:graph-sum-intrinsic-meaning}
The expression \(L^\eta_{\lambda,\omega}(\mathcal H)\) denotes the resulting
intrinsic composite modulo ribbon graphical relations, not a preferred planar
drawing.  In particular, the orientation of a ribbon determines whether its
label is \(X\) or \(X^*\); its passage through a dotted-handle cross-section
does not recolour it.
\end{remark}
