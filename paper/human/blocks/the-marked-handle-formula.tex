\subsection{The marked handle evaluator}
\label{subsec:graphical-construction}

\begin{definition}[Raw decorated Kirby morphism]
\label{def:decorated-kirby-graph}
\label{def:dotted-circle-translation}
\label{def:colored-graph-sum}
Assume the following data.
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{Categorical input:}
  \begin{enumerate}[label=\textup{(\alph*)}]
  \item a premodular fusion category \(\cC\) over \(\mathbb C\);
  \item a monoidal natural automorphism
    \(\eta\colon\operatorname{Id}_{\cC}\Rightarrow
    \operatorname{Id}_{\cC}\).
  \end{enumerate}
\item \emph{Marked handle input:}
  \begin{enumerate}[label=\textup{(\alph*)}]
  \item a one-ended Kirby presentation \(\mathcal H\), whose dotted--framed
    link lies in the standard \(S^3=\partial B^4\);
  \item a handle cocycle \(\omega\in Z^2(\mathcal H)\).
  \end{enumerate}
\item \emph{Label state:} a simple object
  \(\lambda_k\in\operatorname{Irr}(\cC)\) for every framed component
  \(K_k\).
\item \emph{Auxiliary drawing datum} \(\mathfrak a\), consisting of
  \begin{enumerate}[label=\textup{(\alph*)}]
  \item orientations of the framed and dotted components;
  \item an admissible section system for the dotted components;
  \item composition-dual bases used to expand each tensor-unit projector;
  \item a point at infinity outside the resulting graph and a generic planar
    projection of its \(\mathbb R^3\)-model.
  \end{enumerate}
\end{enumerate}

Write \(e_k\) for the oriented \(2\)-handle generator selected by
\(\mathfrak a\), and put
\[
  \omega_k:=\omega(e_k)\in\mathbb Z.
\]
The datum above first produces a closed coloured ribbon diagram
\[
  G^\eta_{\lambda,\omega}(\mathcal H;\mathfrak a).
\]
The component \(K_k\) is coloured by \(\lambda_k\) and carries the coupon
\[
  (\eta^{\omega_k})_{\lambda_k}
  =\eta_{\lambda_k}^{\omega_k}\operatorname{id}_{\lambda_k}.
\]
Here \(\eta^{\omega_k}\) is the integer power of \(\eta\), using its inverse
when \(\omega_k<0\), and
\(\eta_{\lambda_k}\in\mathbb C^\times\) is the simple-object scalar of
\cref{lem:simple-object-character-scalars}.

A dotted component \(\delta\) is not coloured.  Instead, cut its handle as in
\cref{def:cut-open-dotted-tangle}.  If \(B_\delta\) is the tensor product
specified by the ordered oriented strands on one cut face, insert the
canonical projector
\[
  P_{\mathbf1,B_\delta}\colon B_\delta\longrightarrow B_\delta
\]
and reglue the two faces.  Doing this at every dotted component completes
\(G^\eta_{\lambda,\omega}(\mathcal H;\mathfrak a)\).  Interpreting this
closed diagram in \(\cC\) gives
\[
  L^\eta_{\lambda,\omega}(\mathcal H;\mathfrak a)
  :=\bigl[\!\bigl[ G^\eta_{\lambda,\omega}
       (\mathcal H;\mathfrak a)\bigr]\!\bigr]_{\cC}
  \in\operatorname{End}_{\cC}(\mathbf1).
\]
Thus \(L^\eta_{\lambda,\omega}(\mathcal H;\mathfrak a)\) is a morphism,
not a planar drawing.  The Kirby-colour coefficient
\(\prod_k d_{\lambda_k}\) is not included in this notation.
\end{definition}

\begin{proposition}[Independence of the auxiliary drawing datum]
\label{prop:graphical-auxiliary-independence}
Fix \(\cC,\eta,\mathcal H,\omega\), and a label state \(\lambda\) as in
\cref{def:decorated-kirby-graph}.  The endomorphism
\[
  L^\eta_{\lambda,\omega}(\mathcal H;\mathfrak a)
  \in\operatorname{End}_{\cC}(\mathbf1)
\]
is independent of the composition-dual bases, the admissible section system,
the point at infinity, and the regular planar projection.  Reversing an
auxiliary orientation has the same effect as replacing the corresponding
oriented generator, cocycle coordinate, and simple label by
\[
  (e_k,\omega_k,\lambda_k)
  \longmapsto(-e_k,-\omega_k,\lambda_k^*),
\]
and leaves the resulting endomorphism unchanged under the canonical
orientation--duality identification.
\end{proposition}

\begin{proof}
The sum \(\sum_\mu u_\mu\widehat u_\mu\) is the canonical projector onto the
full tensor-unit isotypic summand, so it is independent of the chosen
composition-dual bases.

Any two admissible section systems for the dotted unlink are connected by an
isotopy relative to their boundary circles.  Put the isotopy in generic
position with respect to the compact framed link.  Its finitely many
exceptional parameters are adjacent interchanges, births or deaths of a dual
pair at a cup or cap, and passages across crossings, twists, or coupons.
Projector naturality, in the form of
\cref{cor:finite-unit-projector-transport}, identifies the inserted
endomorphisms across each event.  Induction over the finite event list proves
independence of the section system.  Disjoint dotted-handle neighborhoods
give tensor-independent insertions.

Changing the point at infinity or the regular projection changes only the
drawing of the same embedded ribbon graph.  Ribbon graphical coherence
\cite[Theorem~6.1]{Shum1994} therefore preserves its endomorphism.

Finally, reverse one auxiliary component orientation.  For a framed
component, rigid duality replaces \(\lambda_k\) by \(\lambda_k^*\), while
\(e_k\) and \(\omega_k\) both change sign.  Since
\(d_{\lambda_k^*}=d_{\lambda_k}\) and
\(\eta_{\lambda_k^*}=\eta_{\lambda_k}^{-1}\), the dualized coupon represents
the same oriented morphism.  For a dotted component, the boundary word is
replaced by its reversed rigid dual; pivotal duality carries the canonical
tensor-unit projector to the canonical projector of the dual boundary
object.  Spherical graphical evaluation is unchanged.
\end{proof}

\begin{definition}[Intrinsic decorated Kirby morphism]
\label{def:intrinsic-decorated-kirby-morphism}
By \cref{prop:graphical-auxiliary-independence}, the common value
\[
  L^\eta_{\lambda,\omega}(\mathcal H)
  \in\operatorname{End}_{\cC}(\mathbf1)
\]
is well defined without recording \(\mathfrak a\).  Since the tensor unit is
simple, the canonical identification
\(\operatorname{End}_{\cC}(\mathbf1)\cong\mathbb C\) also assigns a scalar to
this endomorphism; we denote that scalar by
\(\langle L^\eta_{\lambda,\omega}(\mathcal H)\rangle\).
\end{definition}

\begin{definition}[Homologically twisted Crane--Yetter evaluator]
\label{def:twisted-handle-evaluator}
Assume the following data.
\begin{enumerate}[label=\textup{(\roman*)}]
\item a premodular fusion category \(\cC\) over \(\mathbb C\);
\item a monoidal natural automorphism
  \(\eta\colon\operatorname{Id}_{\cC}\Rightarrow
  \operatorname{Id}_{\cC}\);
\item a one-ended Kirby presentation \(\mathcal H\);
\item a handle cocycle \(\omega\in Z^2(\mathcal H)\).
\end{enumerate}
The resulting scalar is
\begin{equation}
  \boxed{
  \CY_{\cC,\eta}(\mathcal H,\omega)
  :=
  \operatorname{Dim}(\cC)^{1-h_3(\mathcal H)}
  \sum_{\lambda}
    \left(\prod_k d_{\lambda_k}\right)
      \bigl\langle L^\eta_{\lambda,\omega}(\mathcal H)\bigr\rangle.}
  \label{eq:twisted-handle-evaluator}
\end{equation}
In this formula:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \(\lambda\) ranges over all assignments of simple objects to the framed
  components;
\item \(d_{\lambda_k}\) is the quantum dimension of the label on \(K_k\);
\item \(\operatorname{Dim}(\cC)=\sum_i d_i^2\) is the global dimension;
\item \(h_3(\mathcal H)\) is the number of \(3\)-handles;
\item \(\langle-\rangle\) is the scalar identification of
  \cref{def:intrinsic-decorated-kirby-morphism}.
\end{enumerate}
\end{definition}

\begin{remark}[Three-dimensional meaning]
\label{rem:planar-projection-after-cutting}
The dotted--framed link and the decorated graph of
\cref{def:decorated-kirby-graph} live in the standard
\(S^3=\partial B^4\).  To draw them, choose a point at infinity disjoint from
the graph and identify its complement with \(\mathbb R^3\), then choose a
generic projection \(\mathbb R^3\to\mathbb R^2\).  Changing either choice
changes only the ribbon diagram, not the embedded coloured ribbon graph.
Ribbon graphical coherence therefore gives the same RT evaluation.
\end{remark}

\begin{proposition}[Dependence on the three-handle data]
\label{prop:evaluator-three-handle-dependence}
Although a one-ended Kirby presentation contains the full attaching embeddings
of its \(3\)-handles, the marked evaluator uses that data only through
\[
  h_3(\mathcal H)
  \quad\text{and}\quad
  \operatorname{im}(\partial_3)\subset C_2(\mathcal H).
\]
More precisely, the scalar in
\eqref{eq:twisted-handle-evaluator} reads \(h_3(\mathcal H)\) in its
normalization, while the requirement that
\(\omega\in Z^2(\mathcal H)\) depends on \(\partial_3\) only through the
condition that \(\omega\) vanish on \(\operatorname{im}(\partial_3)\).
\end{proposition}

\begin{proof}
The coloured ribbon graph and its label sum are constructed entirely from
the dotted--framed \(1\)- and \(2\)-handle diagram.  The displayed formula
contains the \(3\)-handles explicitly only in the exponent
\(1-h_3(\mathcal H)\).  Finally,
\[
  \omega\in Z^2(\mathcal H)
  \quad\Longleftrightarrow\quad
  \omega\circ\partial_3=0,
\]
which depends only on the image of \(\partial_3\).  No other part of the
attaching embeddings enters the formula.
\end{proof}

\begin{proposition}[Finiteness of the evaluator]
\label{prop:evaluator-finiteness}
For a one-ended Kirby datum, the sum in
\eqref{eq:twisted-handle-evaluator} is finite.
\end{proposition}

\begin{proof}
The Kirby-colour sums are finite because \(\cC\) has finitely many simple
isomorphism classes.  Each tensor-unit projector is a finite sum because
\(\cC\) is finite semisimple.  A Kirby diagram has finitely many components,
so only finitely many choices occur.
\end{proof}

\begin{remark}[What the dotted circle is not]
\label{rem:dotted-circle-operation-scope}
A dotted circle is not a coloured ribbon component and is never assigned the
Kirby colour.  It denotes the projector onto the tensor-unit through-channel.
In a modular category that projector admits an alternative encircling
description using \(\operatorname{Dim}(\cC)^{-1}\Omega_{\cC}\), but no such
replacement is used in the premodular construction.
\end{remark}

\begin{remark}[Intrinsic meaning]
\label{rem:graph-sum-intrinsic-meaning}
The expression \(L^\eta_{\lambda,\omega}(\mathcal H)\) denotes the resulting
intrinsic composite modulo ribbon graphical relations, not a preferred planar
drawing.  In particular, the orientation of a ribbon determines whether its
label is \(X\) or \(X^*\); its passage through a dotted-handle cross-section
does not recolour it.
\end{remark}
