\subsection{The formula}
\label{subsec:graphical-construction}

\begin{definition}[Twisted categorical input]
\label{def:twisted-categorical-input}
A twisted categorical input is a pair \((\cC,\eta)\), where \(\cC\) is a
premodular fusion category over \(\mathbb C\) and
\[
  \eta\in A_{\cC}
  =\operatorname{Aut}_{\otimes}(\operatorname{Id}_{\cC}).
\]
\end{definition}

Two standard categorical ingredients enter the formula.

\begin{definition}[Kirby colour and tensor-unit projector]
\label{def:kirby-colour}
\label{def:tensor-unit-projector}
Let \(\cC\) be a premodular fusion category.  Its Kirby colour is
\[
  \Omega_{\cC}:=
  \sum_{i\in\operatorname{Irr}(\cC)}d_i\,i.
  \tag{4.1}\label{eq:kirby-colour}
\]
For an object \(W\), choose a basis
\(u_\mu\colon\mathbf1\to W\) and its composition-dual basis
\(\widehat u_\mu\colon W\to\mathbf1\), and set
\[
  P_{\mathbf1,W}:=
  \sum_\mu u_\mu\widehat u_\mu
  \in\operatorname{End}_{\cC}(W).
  \tag{4.2}\label{eq:tensor-unit-projector}
\]
We also write
\(\operatorname{Cut}_{\cC}(W):=P_{\mathbf1,W}\); no global-dimension factor is
included in this notation.
\end{definition}

\begin{lemma}[Basis independence of the tensor-unit projector]
\label{lem:tensor-unit-projector-basis-independence}
Let \(\cC\) be a premodular fusion category and let \(W\in\cC\).
The endomorphism \(P_{\mathbf1,W}\) in
\eqref{eq:tensor-unit-projector} is independent of the chosen pair of dual
bases.
\end{lemma}

\begin{proof}
If \(v_\nu=\sum_\mu A_{\nu\mu}u_\mu\), then its dual basis is
\(\widehat v_\nu=\sum_\rho(A^{-1})_{\rho\nu}\widehat u_\rho\).  Hence
\[
  \sum_\nu v_\nu\widehat v_\nu
  =\sum_{\mu,\rho}
    \left(\sum_\nu A_{\nu\mu}(A^{-1})_{\rho\nu}\right)
    u_\mu\widehat u_\rho
  =\sum_\mu u_\mu\widehat u_\mu.
\]
\end{proof}

\begin{definition}[The operation represented by a dotted circle]
\label{def:dotted-circle-translation}
Let \(\cC\) be a premodular fusion category, let \(\mathcal H\) be a special
Kirby presentation, and let \(\delta\) be a dotted component of
\(D_{\mathcal H}\).  Fix an assignment \(\lambda\) of simple labels to the
framed components.  A standard
cross-section of the corresponding \(1\)-handle meets the oriented framed
ribbons in a finite ordered boundary word.  If \(B_\delta\) is the tensor
product represented by that word, then the operation assigned to \(\delta\)
is
\[
  \operatorname{Cut}_{\cC}(B_\delta)
  =P_{\mathbf1,B_\delta}.
  \tag{4.3}\label{eq:dotted-operation}
\]
Equivalently, after fixing the exterior ribbon tangle, closure gives a linear
functional
\[
  \mathcal E_\delta\colon
  \operatorname{End}_{\cC}(B_\delta)\longrightarrow\mathbb C,
\]
and the dotted contribution is
\[
  \mathcal E_\delta(P_{\mathbf1,B_\delta}).
  \tag{4.4}\label{eq:intrinsic-dotted-context}
\]
\end{definition}

\begin{convention}[Suppressed label-state dependence]
\label{conv:suppressed-label-state-dependence}
Here and below, the dependence of \(B_\delta\) and \(\mathcal E_\delta\) on
the currently fixed label state \(\lambda\) is suppressed from the notation.
\end{convention}

We can now give the entire evaluation recipe.

\begin{definition}[The coloured graph sum]
\label{def:colored-graph-sum}
Fix a twisted categorical input \((\cC,\eta)\).  Let \(\mathcal H\) be a
special Kirby presentation and let
\(a\in Z^2(\mathcal H)\) represent the marking.  For every assignment
\(\lambda=(\lambda_k)\) of simple objects to the framed components, form a
finite linear combination \(\Gamma(\mathcal H,a;\lambda)\) of closed coloured
ribbon graphs by the following rules.
\begin{enumerate}[label=(\roman*),ref=\roman*]
  \item The \(k\)th framed component receives the simple label
  \(\lambda_k\), with external Kirby-colour coefficient
  \(d_{\lambda_k}\).

  \item On that component insert the character coupon
  \[
    (\eta^{a_k})_{\lambda_k}
    =\eta_{\lambda_k}^{a_k}\operatorname{id}_{\lambda_k}.
  \]

  \item At every dotted component \(\delta\), insert
  \(P_{\mathbf1,B_\delta}\) on the through-channel described in
  \cref{def:dotted-circle-translation}.

  \item Expand the projectors in dual bases, reglue the cut-open local
  tangles, and interpret every crossing, cup, cap, twist, and coupon by the
  ribbon graphical calculus.

  \item Include all projector-expansion coefficients and character coupons
  in \(\Gamma(\mathcal H,a;\lambda)\), but leave the Kirby-colour weight
  \(\prod_kd_{\lambda_k}\) outside this graph sum.
\end{enumerate}
\end{definition}

\begin{definition}[Homologically twisted Crane--Yetter evaluator]
\label{def:twisted-handle-evaluator}
Fix a twisted categorical input \((\cC,\eta)\), a special Kirby presentation
\(\mathcal H\), and a cocycle \(a\in Z^2(\mathcal H)\).  Define
\[
  \boxed{
  \CY_{\cC,\eta}(\mathcal H,a)
  :=
  \operatorname{Dim}(\cC)^{1-h_3(\mathcal H)}
  \sum_{\lambda}
    \left(\prod_k d_{\lambda_k}\right)
    \operatorname{RT}_{\cC}
      \bigl(S^3,\Gamma(\mathcal H,a;\lambda)\bigr).}
  \tag{4.5}\label{eq:twisted-handle-evaluator}
\]
\end{definition}

\begin{proposition}[Finiteness of the evaluator]
\label{prop:evaluator-finiteness}
The sum in \eqref{eq:twisted-handle-evaluator} is finite.
\end{proposition}

\begin{proof}
The Kirby-colour sums are finite because \(\cC\) has finitely many simple
isomorphism classes.  Each tensor-unit projector is a finite sum because
\(\cC\) is finite semisimple.  A Kirby diagram has finitely many components,
so only finitely many choices occur.
\end{proof}

\begin{remark}[What the dotted circle is not]
\label{rem:dotted-circle-operation-scope}
A dotted circle is not a coloured ribbon component and is never assigned the
Kirby colour.  It denotes the projector onto the tensor-unit through-channel.
In a modular category that projector admits an alternative encircling
description using \(\operatorname{Dim}(\cC)^{-1}\Omega_{\cC}\), but no such
replacement is used in the premodular construction; compare
\cite[Definition~5.11 and Lemma~2.31]{Tham2021}.
\end{remark}

\begin{remark}[Intrinsic meaning]
\label{rem:graph-sum-intrinsic-meaning}
The expression \(\Gamma(\mathcal H,a;\lambda)\) denotes the resulting
intrinsic composite modulo ribbon graphical relations, not a preferred planar
drawing.  In particular, the orientation of a ribbon determines whether its
label is \(X\) or \(X^*\); its passage through a dotted-handle cross-section
does not recolour it.
\end{remark}
