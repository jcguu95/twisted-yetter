\subsection{From the untwisted handle morphism to the homological twist}
\label{subsec:graphical-construction}

\begin{definition}[Untwisted premodular handle morphism]
\label{def:untwisted-handle-morphism}
\label{def:dotted-circle-translation}
Assume the following data.
\begin{enumerate}[label=\textup{(\roman*)}]
\item a premodular fusion category \(\cC\) over \(\mathbb C\);
\item a one-ended Kirby presentation \(\mathcal H\), whose dotted--framed
  link lies in the standard \(S^3=\partial B^4\);
\item a simple object \(\lambda_k\in\operatorname{Irr}(\cC)\) for every
  framed component \(K_k\);
\item auxiliary orientations, an admissible dotted-handle section system,
  composition-dual bases for the tensor-unit projectors, and a regular planar
  drawing; denote this collection by \(\mathfrak a\).
\end{enumerate}
The framed component \(K_k\) is coloured by \(\lambda_k\), with no additional
coupon.  A dotted component \(\delta\) is not coloured: cut its handle, insert
\[
  P_{\mathbf1,B_\delta}\colon B_\delta\longrightarrow B_\delta
\]
on the ordered boundary object, and reglue.  This gives a closed ribbon
diagram \(G_\lambda(\mathcal H;\mathfrak a)\) and hence
\[
  L_\lambda(\mathcal H;\mathfrak a)
  :=\bigl[\!\bigl[G_\lambda(\mathcal H;\mathfrak a)\bigr]\!\bigr]_{\cC}
  \in\operatorname{End}_{\cC}(\mathbf1).
\]
The Kirby-colour coefficient \(\prod_kd_{\lambda_k}\) is not included in
this notation.
\end{definition}

\begin{proposition}[Untwisted auxiliary-choice independence]
\label{prop:untwisted-auxiliary-independence}
The family \(L_\lambda(\mathcal H;\mathfrak a)\) is independent of the
projector bases, section system, point at infinity, and planar projection.
Reversing an auxiliary component orientation and replacing its label by the
dual label gives the same endomorphism under the canonical
orientation--duality identification.
\end{proposition}

\begin{proof}
Composition-dual resolutions give the canonical tensor-unit projector and
are therefore basis independent.  A generic isotopy between two admissible
section systems meets the compact framed link in finitely many elementary
ribbon events; projector naturality across each event and induction give the
same reglued morphism.  Ribbon coherence
\cite[Theorem~6.1]{Shum1994} removes the choices of point at infinity and
planar projection.  Finally, rigid duality replaces a reversed framed label
by its dual and carries a reversed dotted boundary word, together with its
tensor-unit projector, to the pivotal transpose.  Spherical evaluation is
unchanged.
\end{proof}

\begin{definition}[Intrinsic untwisted handle morphism]
\label{def:intrinsic-untwisted-handle-morphism}
The preceding proposition defines
\[
  L_\lambda(\mathcal H)\in\operatorname{End}_{\cC}(\mathbf1)
\]
without recording \(\mathfrak a\), up to the canonical dual reindexing of
label states.
\end{definition}

\begin{definition}[Untwisted premodular handle evaluator]
\label{def:untwisted-handle-evaluator}
The untwisted value of a one-ended Kirby presentation is
\[
  \CY^{\mathrm{un}}_{\cC}(\mathcal H)
  :=
  \operatorname{Dim}(\cC)^{1-h_3(\mathcal H)}
  \sum_\lambda
    \left(\prod_kd_{\lambda_k}\right)
    \bigl\langle L_\lambda(\mathcal H)\bigr\rangle,
\]
where \(\langle-\rangle\) is the canonical scalar identification
\(\operatorname{End}_{\cC}(\mathbf1)\cong\mathbb C\).
\end{definition}

\begin{definition}[Homological character decoration]
\label{def:decorated-kirby-graph}
In addition, let
\[
  \eta\colon\operatorname{Id}_{\cC}\Rightarrow\operatorname{Id}_{\cC}
\]
be a monoidal natural automorphism and let
\(\omega\in Z^2(\mathcal H)\).  If \(e_k\) is the oriented \(2\)-handle
generator selected by \(\mathfrak a\), put
\(\omega_k:=\omega(e_k)\in\mathbb Z\).  Starting from the untwisted diagram,
place on \(K_k\) the single coupon
\[
  (\eta^{\omega_k})_{\lambda_k}
  =\eta_{\lambda_k}^{\omega_k}\operatorname{id}_{\lambda_k},
\]
where \(\eta_{\lambda_k}\in\mathbb C^\times\) is the scalar of
\cref{lem:simple-object-character-scalars}.  Denote the resulting diagram by
\(G^\eta_{\lambda,\omega}(\mathcal H;\mathfrak a)\), and set
\[
  L^\eta_{\lambda,\omega}(\mathcal H;\mathfrak a)
  :=\bigl[\!\bigl[G^\eta_{\lambda,\omega}
       (\mathcal H;\mathfrak a)\bigr]\!\bigr]_{\cC}
  \in\operatorname{End}_{\cC}(\mathbf1).
\]
This coupon is the only modification: every dotted component still inserts
the same tensor-unit projector \(P_{\mathbf1,B_\delta}\).
\end{definition}

\begin{proposition}[The twist preserves auxiliary-choice independence]
\label{prop:graphical-auxiliary-independence}
The endomorphism family
\(L^\eta_{\lambda,\omega}(\mathcal H;\mathfrak a)\) has the same
auxiliary-choice independence as the untwisted family.  Under orientation
reversal the precise reindexing is
\[
  (e_k,\omega_k,\lambda_k)
  \longmapsto(-e_k,-\omega_k,\lambda_k^*).
\]
\end{proposition}

\begin{proof}
Only the character coupons are new.  Naturality transports them across every
ribbon event, monoidality handles tensor products, and
\(\eta_{\mathbf1}=\operatorname{id}_{\mathbf1}\) leaves the tensor-unit
projectors unchanged.  Under orientation reversal,
\(\eta_{X^*}=\eta_X^{-1}\) identifies the dualized coupon after both the
generator and its cocycle coordinate change sign.  These are exactly the
additional checks beyond
\cref{prop:untwisted-auxiliary-independence}.
\end{proof}

\begin{definition}[Intrinsic decorated Kirby morphism]
\label{def:intrinsic-decorated-kirby-morphism}
The common value
\[
  L^\eta_{\lambda,\omega}(\mathcal H)
  \in\operatorname{End}_{\cC}(\mathbf1)
\]
is well defined without \(\mathfrak a\).  Under
\(\operatorname{End}_{\cC}(\mathbf1)\cong\mathbb C\), denote its scalar by
\(\langle L^\eta_{\lambda,\omega}(\mathcal H)\rangle\).
\end{definition}

\begin{definition}[Homologically twisted Crane--Yetter evaluator]
\label{def:twisted-handle-evaluator}
Assume the following data.
\begin{enumerate}[label=\textup{(\roman*)}]
\item a premodular fusion category \(\cC\) over \(\mathbb C\);
\item a monoidal natural automorphism
  \(\eta\colon\operatorname{Id}_{\cC}\Rightarrow
  \operatorname{Id}_{\cC}\);
\item a one-ended Kirby presentation \(\mathcal H\);
\item a handle cocycle \(\omega\in Z^2(\mathcal H)\).
\end{enumerate}
The resulting scalar is
\begin{equation}
  \boxed{
  \CY_{\cC,\eta}(\mathcal H,\omega)
  :=
  \operatorname{Dim}(\cC)^{1-h_3(\mathcal H)}
  \sum_{\lambda}
    \left(\prod_k d_{\lambda_k}\right)
      \bigl\langle L^\eta_{\lambda,\omega}(\mathcal H)\bigr\rangle.}
  \label{eq:twisted-handle-evaluator}
\end{equation}
In this formula:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \(\lambda\) ranges over all assignments of simple objects to the framed
  components;
\item \(d_{\lambda_k}\) is the quantum dimension of the label on \(K_k\);
\item \(\operatorname{Dim}(\cC)=\sum_i d_i^2\) is the global dimension;
\item \(h_3(\mathcal H)\) is the number of \(3\)-handles;
\item \(\langle-\rangle\) is the scalar identification of
  \cref{def:intrinsic-decorated-kirby-morphism}.
\end{enumerate}
\end{definition}

\begin{proposition}[Exact untwisted specialization]
\label{prop:exact-untwisted-specialization}
For every \(\mathcal H,\lambda\), and \(\omega\),
\[
  L^{\operatorname{id}}_{\lambda,\omega}(\mathcal H)
  =L_\lambda(\mathcal H),
  \qquad
  L^\eta_{\lambda,0}(\mathcal H)=L_\lambda(\mathcal H).
\]
Consequently,
\[
  \CY_{\cC,\operatorname{id}}(\mathcal H,\omega)
  =\CY^{\mathrm{un}}_{\cC}(\mathcal H),
  \qquad
  \CY_{\cC,\eta}(\mathcal H,0)
  =\CY^{\mathrm{un}}_{\cC}(\mathcal H).
\]
\end{proposition}

\begin{proof}
In either case every character coupon is an identity; all remaining labels,
projectors, and regluing operations are exactly the untwisted ones.
\end{proof}

\begin{remark}[Three-dimensional meaning]
\label{rem:planar-projection-after-cutting}
The dotted--framed link and the decorated graph of
\cref{def:decorated-kirby-graph} live in the standard
\(S^3=\partial B^4\).  To draw them, choose a point at infinity disjoint from
the graph and identify its complement with \(\mathbb R^3\), then choose a
generic projection \(\mathbb R^3\to\mathbb R^2\).  Changing either choice
changes only the ribbon diagram, not the embedded coloured ribbon graph.
Ribbon graphical coherence therefore gives the same RT evaluation.
\end{remark}

\begin{proposition}[Dependence on the three-handle data]
\label{prop:evaluator-three-handle-dependence}
Although a one-ended Kirby presentation contains the full attaching embeddings
of its \(3\)-handles, the marked evaluator uses that data only through
\[
  h_3(\mathcal H)
  \quad\text{and}\quad
  \operatorname{im}(\partial_3)\subset C_2(\mathcal H).
\]
More precisely, the scalar in
\eqref{eq:twisted-handle-evaluator} reads \(h_3(\mathcal H)\) in its
normalization, while the requirement that
\(\omega\in Z^2(\mathcal H)\) depends on \(\partial_3\) only through the
condition that \(\omega\) vanish on \(\operatorname{im}(\partial_3)\).
\end{proposition}

\begin{proof}
The coloured ribbon graph and its label sum are constructed entirely from
the dotted--framed \(1\)- and \(2\)-handle diagram.  The displayed formula
contains the \(3\)-handles explicitly only in the exponent
\(1-h_3(\mathcal H)\).  Finally,
\[
  \omega\in Z^2(\mathcal H)
  \quad\Longleftrightarrow\quad
  \omega\circ\partial_3=0,
\]
which depends only on the image of \(\partial_3\).  No other part of the
attaching embeddings enters the formula.
\end{proof}

\begin{proposition}[Finiteness of the evaluator]
\label{prop:evaluator-finiteness}
For a one-ended Kirby datum, the sum in
\eqref{eq:twisted-handle-evaluator} is finite.
\end{proposition}

\begin{proof}
The Kirby-colour sums are finite because \(\cC\) has finitely many simple
isomorphism classes.  Each tensor-unit projector is a finite sum because
\(\cC\) is finite semisimple.  A Kirby diagram has finitely many components,
so only finitely many choices occur.
\end{proof}

\begin{remark}[What the dotted circle is not]
\label{rem:dotted-circle-operation-scope}
A dotted circle is not a coloured ribbon component and is never assigned the
Kirby colour.  It denotes the projector onto the tensor-unit through-channel.
In a modular category that projector admits an alternative encircling
description using \(\operatorname{Dim}(\cC)^{-1}\Omega_{\cC}\), but no such
replacement is used in the premodular construction.
\end{remark}

\begin{remark}[Intrinsic meaning]
\label{rem:graph-sum-intrinsic-meaning}
The expression \(L^\eta_{\lambda,\omega}(\mathcal H)\) denotes the resulting
intrinsic composite modulo ribbon graphical relations, not a preferred planar
drawing.  In particular, the orientation of a ribbon determines whether its
label is \(X\) or \(X^*\); its passage through a dotted-handle cross-section
does not recolour it.
\end{remark}
