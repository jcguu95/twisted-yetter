\subsection{Topological input}
\label{subsec:topological-input}

\begin{definition}[Marked four-manifold]
\label{def:marked-four-manifold}
A \emph{marked four-manifold} is a pair $(X,\alpha)$ consisting of
\begin{enumerate}[label=(\roman*),ref=\roman*]
  \item\label{item:marked-manifold}
  a closed, connected, oriented smooth four-manifold $X$; and
  \item\label{item:manifold-marking}
  a class $\alpha\in H_2(X;\mathbb Z)$.
\end{enumerate}
This is our fixed-character integral specialization of Yetter's general
coefficient-group programme \cite[pp.~2--3]{Yetter1993}.  More precisely,
fixing $\eta\in A_{\cC}$ gives a coefficient homomorphism
$\mathbb Z\to A_{\cC}$, $n\mapsto\eta^n$.  Yetter's four-dimensional state
sum in Definition~6 and its invariance in Theorem~8 are instead written for
$H_2(-;\mathbb Z_2)$; neither theorem is being invoked for the integral
specialization defined in this paper.  Representative independence and
Kirby-move invariance are therefore proved below.  Yetter's Definition~9
packages the marked values of his displayed theory into a total invariant;
that is also not the definition being used here.
\end{definition}

\begin{definition}[Special Kirby representative]
\label{def:special-kirby-representative}
A \emph{special Kirby representative} of $(X,\alpha)$ consists of
\begin{enumerate}[label=(\roman*),ref=\roman*]
  \item\label{item:special-handles}
  a handle decomposition of $X$ with one $0$-handle and one $4$-handle;

  \item\label{item:special-diagram}
  its special framed link $D=D_{\mathrm{dot}}\sqcup D_{\mathrm{fr}}$, where
  $D_{\mathrm{dot}}$ is the standard dotted unlink, its dotted circles
  represent primal $1$-handles, and
  the integer-framed circles in $D_{\mathrm{fr}}$ represent primal
  $2$-handles, together with auxiliary orientations of all link components;

  \item\label{item:special-oriented-bases}
  oriented generators of the primal $1$- and $2$-handles, with the link
  orientation of each component tied to its handle generator;

  \item\label{item:special-three-handles}
  the number $h_3(D)$ of primal $3$-handles and the image of their algebraic
  boundary map in $C_2(D)$.  A basis of $C_3(D)$ may be chosen to write the
  map, but is not part of the representative: two matrices differing by an
  integral change of $C_3$-basis record the same data.  We do not require a
  computational encoding of each attaching $2$-sphere;

  \item\label{item:special-closing}
  the condition that the recorded $3$-handles and the unique $4$-handle cap
  the boundary remaining after the $2$-handles are attached; and

  \item\label{item:special-marking}
  the class in handle cohomology corresponding to the Poincar\'e dual
  $\operatorname{PD}_X(\alpha)$.
\end{enumerate}
The dotted-circle convention is reviewed in
\cite[Section~5.4]{GompfStipsicz1999} and
\cite[Section~2.3.2]{BarenzBarrett2018}.  The individual attaching maps of
the $3$- and $4$-handles are omitted because the Kirby diagram records the
$0$-, $1$-, and $2$-handlebody together with the assertion that the remaining
boundary is capped by the stated $3$- and $4$-handles; this is the usual
Kirby-calculus convention of \cite[Sections~4.2 and~5.4]{GompfStipsicz1999}.
The component orientations and oriented handle generators are auxiliary.
Their independence is proved in
\cref{prop:orientation-choice-independence}; an unoriented Kirby diagram does
not acquire additional topological structure from these choices.
\end{definition}

\begin{definition}[Handle cohomology and marking]
\label{def:handle-homology-marking}
For a special Kirby diagram $D$, let
\[
  C_3(D)\xrightarrow{\partial_3}
  C_2(D)\xrightarrow{\partial_2}C_1(D),
  \qquad \partial_2\partial_3=0,
\]
be the integral handle chain complex, with $C_1(D)$ and $C_2(D)$ freely
generated by the oriented primal handles.  A basis of $C_3(D)$ is used only
to display $\partial_3$; changing it by
$g\in\operatorname{GL}(C_3(D);\mathbb Z)$ replaces $\partial_3$ by
$\partial_3g$ and leaves $\operatorname{im}(\partial_3)$ unchanged.  Its
dual cochain complex in the relevant degrees is
\[
  C^1(D)\xrightarrow{\delta^1=\partial_2^*}
  C^2(D)\xrightarrow{\delta^2=\partial_3^*}C^3(D),
  \qquad \delta^2\delta^1=0,
\]
and
\[
  H^2(D):=\ker(\delta^2)/\operatorname{im}(\delta^1).
\]
The diagrammatic marking associated to
$\alpha\in H_2(X;\mathbb Z)$ is a class
\[
  \xi_D\in H^2(D)
\]
which corresponds to $\operatorname{PD}_X(\alpha)$ under the cellular
comparison isomorphism.  A cocycle
$a\in\ker(\delta^2)\subset C^2(D)$ representing $\xi_D$ may be chosen only
temporarily when writing the state sum.  Thus the exponent $a_k$ is the
value of this cellular $2$-cochain on the $k$th primal $2$-handle; it is not
the coefficient of a primal handle $2$-cycle.  This is the cellular cochain
complex dual to the handle complex; see the discussion following
\cite[Proposition~4.2.13]{GompfStipsicz1999}.
\end{definition}

\begin{lemma}[$C_3$-basis independence]
\label{lem:c3-basis-independence}
The group $H^2(D)$, its marked class, and the number $h_3(D)$ are unchanged
by an integral change of basis in $C_3(D)$.
\end{lemma}

\begin{proof}
If $g\in\operatorname{GL}(C_3(D);\mathbb Z)$, then the new cochain map is
$g^*\partial_3^*$.  Since $g^*$ is invertible,
$\ker(g^*\partial_3^*)=\ker(\partial_3^*)$.  Neither $C^1$, $C^2$ nor
$\operatorname{im}(\delta^1)$ changes, so the quotient defining $H^2(D)$
is identical.  The cellular comparison is natural for this based chain
isomorphism, and a basis change does not alter the rank of $C_3$.
\end{proof}

\begin{definition}[Dotted co-core chart]
\label{def:dotted-cut-system}
A \emph{dotted co-core chart} for $D$ is obtained from the actual handle
decomposition carried by the special Kirby representative, as follows.  For
each dotted component $\delta$, choose standard pair-of-balls coordinates on
the corresponding primal $1$-handle and turn those coordinates over to the
dual index-$3$ handle.  The choices for distinct handles have disjoint
supports.  A generic ribbon-diagram projection of these fixed local handle
models consists of
\begin{enumerate}[label=(\roman*),ref=\roman*]
\item\label{item:cut-disks}
  the cooriented spanning disk that displays each chosen pair-of-balls model,
  with coorientation tied to the oriented primal $1$-handle generator.  The
  displaying disks have pairwise disjoint product neighborhoods and are
  transverse to the framed $2$-handle link.  Write
  $\Delta_\delta\times[-1,1]$ for the product collar, and put
  $\Delta_\delta^+:=\Delta_\delta\times\{+1\}$ for the face selected by the
  coorientation;

  \item\label{item:cut-punctures}
  the finite set of punctures created on each disk by the transverse
  intersections.  A puncture $p$ records its ordinary component $k(p)$,
  the intersection sign $\epsilon(p)\in\{+1,-1\}$, and the orientation of
  the ribbon strand.  The sign is positive precisely when the oriented
  strand crosses the disk in the positive normal direction, and it is the
  local contribution to the coefficient of the corresponding $1$-handle in
  $\partial_2(e_{k(p)})$;

  \item\label{item:cut-order}
  a local parameterization which reads the punctures
  $p_1,\ldots,p_r$ in order on the positive collar face.  The boundary
  occurrence at $p_j$ has variance
  $\sigma_j:=\epsilon(p_j)$.  On the opposite face the order is reversed and
  every variance is dualized.  The oriented ribbon label is fixed before the
  disk is chosen; variance records how that already coloured ribbon occurs in
  a boundary-state object and does not recolour the component; and

  \item\label{item:cut-regluing}
  the coordinate regluing that reconstructs the original local handle models
  and hence the original Kirby diagram.
\end{enumerate}
For later use, define the signed incidence number
\[
  I_\delta(k)
  :=\sum_{j:\,k(p_j)=k}\epsilon(p_j)\in\mathbb Z.
  \tag{2.1}\label{eq:dotted-incidence-number}
\]
If $e_k\in C_2(D)$ is the generator represented by the $k$th ordinary
component and $e_\delta\in C_1(D)$ is the generator represented by the
dotted component $\delta$, then the cellular attaching map is
\[
  \partial_2(e_k)=\sum_\delta I_\delta(k)e_\delta.
  \tag{2.2}\label{eq:dotted-cellular-boundary}
\]
Thus the ordered word retains the full cut-open boundary state, whereas
$I_\delta(k)$ is only its algebraic incidence count.  The former is used by
the graphical operator and the latter by the handle cochain complex.
The puncture order in \cref{item:cut-order} is part of the cut-open tangle;
it is not replaced by an unordered incidence count.
This is not an arbitrary spanning-disk system.  It is a graphical coordinate
chart on the actual handles already present in
\cref{def:special-kirby-representative}.  It is neither additional evaluator
input nor a second realization of the Kirby diagram.
The positive face and the opposite-face reversal agree with the pair-of-balls
description of a $1$-handle in
\cite[Section~2.3.2 and Definition~4.12]{BarenzBarrett2018}.  Its relation to
the turned-over index-$3$ operation is proved in
\cref{prop:dotted-chart-realization}; the definition itself does not assume an
evaluation equality.
\end{definition}

\begin{lemma}[Existence of dotted co-core charts]
\label{lem:dotted-cut-presentation-existence}
Every special Kirby representative admits a dotted co-core chart.
\end{lemma}

\begin{proof}
The handle decomposition already supplies pairwise disjoint local models for
its $1$-handles.  Choose standard pair-of-balls coordinates in those models.
In dotted-circle notation their displaying disks form the standard dotted
unlink.  After a perturbation supported inside the chosen coordinate
neighborhoods, the disks are transverse to the compact framed $2$-handle link
and hence have finitely many punctures.  The coordinates supply the product
collars, the ordered faces, and the regluing maps.
\end{proof}

\begin{proposition}[Co-core charts render the intrinsic index-$3$ maps]
\label{prop:dotted-chart-realization}
Let $\delta$ be a dotted component of a special Kirby representative.
In the standard pair-of-balls model of its primal $1$-handle, the positive
collar face of a dotted co-core chart records the ordered signed ribbon
word carried by the strands traversing that handle.  After turning the handle
decomposition over, this word presents the ribbons made transverse to the
\emph{co-core} of the dual index-$3$ handle, which is the locus used by the
index-$3$ operation of \cite[Definition~5.11, $k=3$]{Tham2021}.  The
two-dimensional collar face is a graphical chart for that word; it is not
being identified with the one-dimensional co-core itself.

Consequently the projector drawn in this chart is a rendering of the fixed
intrinsic index-$3$ handle map, not a map defined by the chart.  For all dotted
components at once, the disjoint handle neighborhoods render the tensor
product of the corresponding intrinsic maps; changing their written order is
only the interchange law for disjoint morphisms.
\end{proposition}

\begin{proof}
Use the standard pair-of-balls model $D^1\times D^3$ for the primal
$1$-handle.  In Akbulut's dotted-circle notation, thickening a spanning disk
for $\delta$ recovers these two attaching $3$-balls; ordinary attaching
circles crossing the disk become the arcs entering one ball and leaving the
other \cite[Section~2.3.2, Figure~1]{BarenzBarrett2018}.  The positive collar
face records those traversing arcs, with their order and variance.  Turning
the handle decomposition over replaces the primal $1$-handle by a dual
index-$3$ handle.  The same traversing ribbons can then be bunched and made
transverse to that handle's co-core.  This is exactly the dotted-circle
calculation used in the proof of
\cite[proof of Proposition~9.4, p.~96]{Tham2021}: removing primal $1$-handles,
or equivalently attaching the dual $3$-handles, applies Definition~5.11 after
the ribbons through each $1$-handle have been bunched using Lemma~2.27.
Thus the collar chart records the typed boundary word of the co-core
transverse graph without identifying the disk face with the co-core.
Definition~5.11, $k=3$, assigns the intrinsic handle map to this actual
co-core; the chart merely draws it.  Since the local handle neighborhoods are
disjoint, their maps tensor independently, and the interchange law compares
different written orders.  No uniqueness theorem for arbitrary systems of
spanning disks is used.
\end{proof}

\begin{convention}[Three independent sources of signs]
\label{conv:three-sign-sources}
The following information must not be conflated.
\begin{enumerate}[label=(\roman*),ref=\roman*]
  \item\label{item:strand-dual-sign}
  The orientation of a ribbon strand determines whether its label is $X$ or
  $X^*$.

  \item\label{item:cut-direction-sign}
  The cooriented disk and the orientation of the attaching circle determine
  the cellular intersection sign $\epsilon(p)$ and the variance of the
  corresponding oriented boundary point in the cut section.

  \item\label{item:character-exponent-source}
  A representative of the cohomology marking supplies the integral character
  exponent attached to each ordinary $2$-handle.
\end{enumerate}
In particular, \cref{item:cut-direction-sign} determines neither
\cref{item:strand-dual-sign} nor \cref{item:character-exponent-source}.
The co-core chart is auxiliary to the proof and is not an additional input to
the manifold-level invariant.
The first convention is the standard orientation-duality convention for
colored ribbon graphs \cite[Chapter~2, Section~2.2]{BakalovKirillov2001};
the third follows the general coefficient-group programme of
\cite[pp.~2--3 and Definitions~1 and~3]{Yetter1993}.  The integral
fixed-character specialization and its invariance are arguments of this
paper, not consequences quoted from Yetter's $\mathbb Z_2$ Theorem~8.
\end{convention}

\begin{convention}[Primal and turned-over handles]
\label{conv:primal-turned-over}
Topological handle names are primal.  The evaluator applies
\cite[Definition~5.11]{Tham2021} to the turned-over decomposition, so
\[
  h^{\mathrm{Tham}}_1=h^{\mathrm{pr}}_3,
  \qquad h^{\mathrm{Tham}}_2=h^{\mathrm{pr}}_2,
  \qquad h^{\mathrm{Tham}}_3=h^{\mathrm{pr}}_1.
\]
Poincar\'e duality and the cellular comparison transport the primal homology
marking to the displayed handle-cohomology marking before a representative
cocycle and its character exponents are chosen.  Turning a handle
decomposition upside down is
described in \cite[Section~4.2]{GompfStipsicz1999}; the evaluator convention
is \cite[Definition~5.11]{Tham2021}.
\end{convention}

\begin{definition}[Elementary Kirby event]
\label{def:kirby-event}
An elementary Kirby event from a special Kirby diagram $D$ to a special
Kirby diagram $D'$ is one of the following eight directed events:
\begin{enumerate}[label=(\roman*),ref=\roman*]
  \item\label{move:framed-isotopy} framed isotopy;
  \item\label{move:one-over-one} a $1/1$ slide;
  \item\label{move:two-over-one} a $2/1$ slide;
  \item\label{move:two-over-two} a $2/2$ slide;
  \item\label{move:one-two-birth} a $1$--$2$ birth;
  \item\label{move:one-two-death} a $1$--$2$ death;
  \item\label{move:two-three-birth} a $2$--$3$ birth; or
  \item\label{move:two-three-death} a $2$--$3$ death.
\end{enumerate}
For an $r/s$ slide, $r$ and $s$ record the roles of the moving and stationary
handles.  Every event includes a chain-homotopy equivalence
\[
  F_m\colon C_*(D)\longrightarrow C_*(D')
\]
given by cellular continuation.  Since cohomology is contravariant, its
pullback is
\[
  F_m^*\colon H^2(D')\longrightarrow H^2(D).
\]
We write
\[
  T_m:=(F_m^*)^{-1}\colon H^2(D)\longrightarrow H^2(D')
\]
for its source-to-target cohomology transport.  The event contains no
marking and no equality of state-sum evaluations.  The list is the directed
form of
\cite[Table~2]{BarenzBarrett2018}; the dotted-circle slides are reviewed in
\cite[Section~5.4]{GompfStipsicz1999}.
\end{definition}

\begin{convention}[Referenced local topology dictionary]
\label{conv:referenced-kirby-dictionary}
The eight names in \cref{def:kirby-event} refer to the following standard
local models.  This dictionary fixes the topological source and target of
each event; it contains no categorical decoration and asserts no equality of
evaluations.
\begin{enumerate}[label=(\roman*),ref=\roman*]
  \item A framed isotopy is an ambient isotopy of the framed link, preserving
  the ordinary/dotted roles of its components.

  \item A $1/1$ slide is the dotted-circle move of
  \cite[Figure~5.39]{GompfStipsicz1999}: one primal $1$-handle is slid over
  another.  Algebraically it is an elementary change of the $C_1$ basis,
  together with the induced rewriting of $\partial_2$; no ordinary
  $2$-handle is created or deleted.

  \item A $2/1$ slide is the move of
  \cite[Figure~5.36]{GompfStipsicz1999}: an attaching arc of a primal
  $2$-handle is pulled through a stationary primal $1$-handle.  The handle
  generators persist, and the source and target attaching maps are related by
  the canonical cellular continuation of this isotopy in the $1$-handlebody.

  \item A $2/2$ slide is the usual oriented band sum of a moving framed
  $2$-handle with a framed push-off of a stationary one
  \cite[Section~5.1]{GompfStipsicz1999}.  Its two oriented basis changes are
  exactly those fixed in \cref{conv:two-two-coordinates}.

  \item A $1$--$2$ birth inserts the isolated dotted circle and its
  $0$-framed meridian shown in
  \cite[Figure~5.38]{GompfStipsicz1999}; on the handle complex it adds the
  acyclic summand $\mathbb Z\xrightarrow{\,\pm1\,}\mathbb Z$ in degrees
  $2\to1$.  A $1$--$2$ death is its inverse after the pair has been isolated.

  \item A $2$--$3$ birth adds the standard cancelling pair and hence the
  acyclic summand $\mathbb Z\xrightarrow{\,\pm1\,}\mathbb Z$ in degrees
  $3\to2$.  A $2$--$3$ death is its inverse.
\end{enumerate}
Apart from framed isotopy, these are the five undirected move/cancellation
rows in \cite[Table~2]{BarenzBarrett2018}, with each cancellation separated
here into its birth and death directions.  In every case the displayed local model
is supported in a ball or a cancelling-pair neighborhood and the framed
tangle outside that neighborhood is identified pointwise.
\end{convention}

\begin{definition}[Marked elementary Kirby event]
\label{def:marked-kirby-event}
A marked elementary Kirby event consists of an elementary event
$m\colon D\to D'$ and classes $\xi_D\in H^2(D)$ and
$\xi_{D'}\in H^2(D')$ such that
\[
  \xi_{D'}=T_m(\xi_D),
  \qquad\text{equivalently}\qquad
  F_m^*(\xi_{D'})=\xi_D.
\]
This adds marking compatibility, not an equality of state-sum evaluations.
\end{definition}

\begin{convention}[Auxiliary orientation reversal]
\label{conv:orientation-reversal}
Reversing an ordinary $2$-handle generator simultaneously reverses the
corresponding ribbon orientation.  In coordinates this sends
\[
  e_k\longmapsto-e_k,\qquad a_k\longmapsto-a_k,
  \qquad i\longmapsto i^*.
\]
Reversing a dotted $1$-handle generator simultaneously reverses the dotted
circle orientation and disk coorientation, so the signed boundary word is
replaced by its rigid dual, with reversed order.  These are changes of
auxiliary oriented bases, not Kirby moves.
\end{convention}

\begin{convention}[$2/2$ coordinates and channel orientation]
\label{conv:two-two-coordinates}
Let $e_x$ and $e_y$ denote the moving and stationary primal $2$-handle
basis elements, and let a cellular $2$-cochain take the values $a$ and $b$
on them.  For the positive slide we use
\[
  e_x'=e_x+e_y,
  \qquad (a',b')=(a+b,b),
\]
and orient the attachment so that its internal simple channels satisfy
$z\subset x\otimes y^*$.  For the negative slide we use
\[
  e_x'=e_x-e_y,
  \qquad (a',b')=(a-b,b),
\]
and orient the attachment so that $z\subset x\otimes y$.  These are
coordinate conventions for the handle-basis change and its dual cochain
map; they do not assert an equality of graphical evaluations.
\end{convention}

The topology used in the descent from diagrams to manifolds is isolated in
the following statements.

\begin{proposition}[Existence of special representatives]
\label{prop:special-representative-existence}
Every marked four-manifold $(X,\alpha)$ admits a special Kirby representative
$(D,\xi_D)$ in which $\xi_D$ represents
$\operatorname{PD}_X(\alpha)$.
\end{proposition}

\begin{proof}
Choose a Morse function with a unique minimum and maximum, pass to its handle
decomposition, and use dotted-framed Kirby notation.  This is
\cite[Proposition~4.2.13]{GompfStipsicz1999} and its dual argument.
\end{proof}

\begin{proposition}[Handle-cohomology realization]
\label{prop:handle-homology-realization}
If $D$ is a special Kirby representative of $X$, there is a natural
isomorphism
\[
  H^2(D)\cong H^2(X;\mathbb Z)
\]
carrying $\xi_D$ to $\operatorname{PD}_X(\alpha)$.
\end{proposition}

\begin{proof}
The handle decomposition is a relative CW decomposition, and its cellular
chain complex is the handle complex in
\cref{def:handle-homology-marking}.  See the handle-homology discussion
immediately following \cite[Proposition~4.2.13]{GompfStipsicz1999}.
\end{proof}

\begin{lemma}[Naturality of the marked handle complex]
\label{lem:marked-handle-complex-naturality}
Every event in \cref{def:kirby-event} determines a chain-homotopy
equivalence of the associated handle complexes: a same-index slide is a based
elementary change in the appropriate chain group, a $2/1$ move is continuation
under an isotopy of a $2$-handle attaching map in the $1$-handlebody, and a
birth or death is an elementary expansion or collapse by an acyclic
adjacent-degree summand.  Under the
cellular comparison maps, the induced map on handle cohomology is the map on
$H^2$ induced by the underlying diffeomorphism of the represented
four-manifolds.  More precisely, if
$F_m\colon C_*(D)\to C_*(D')$ is the cellular continuation map, then the
comparison maps intertwine the contravariant map
$F_m^*\colon H^2(D')\to H^2(D)$ with singular-cohomology pullback.  Thus the
source-to-target marking transport is $T_m=(F_m^*)^{-1}$.
\end{lemma}

\begin{proof}
For a same-index handle slide, replace the moving handle generator by its signed
sum with the stationary generator and leave the other generators fixed; the
boundary matrices change by the corresponding row or column operation, so this
is a chain isomorphism.  A $2/1$ move changes only the chosen attaching-map
drawing through an isotopy in the $1$-handlebody and has its canonical cellular
continuation.  For a cancelling birth, adjoin the two generators whose
only new boundary coefficient is $\pm1$; they form a contractible two-term
summand.  Death is the inverse collapse.  Framed isotopy gives the canonical
identification.  These are the cellular continuation maps of the two relative
CW structures.  Naturality of cellular cohomology with respect to cellular
continuation and its comparison with singular cohomology gives the stated
contravariant commutative square.  Each continuation map is a
chain-homotopy equivalence, so $F_m^*$ is invertible and defines $T_m$.  In
particular, this lemma records marking transport; it does not assert any
equality of graphical evaluations.
\end{proof}

\begin{proposition}[Relative fixed-extrema Cerf movie]
\label{prop:relative-fixed-extrema-cerf}
Let $\mathcal H_0$ and $\mathcal H_1$ be special handle decompositions of the
same closed connected oriented smooth four-manifold.  After isotopy, choose
standard neighborhoods $B_-$ and $B_+$ of their unique minima and maxima.
On
\[
  W=X\setminus\bigl(\operatorname{int}B_-\sqcup
                     \operatorname{int}B_+\bigr),
\]
the two decompositions are joined by a finite relative Cerf movie with the
following properties:
\begin{enumerate}[label=(\roman*),ref=\roman*]
  \item every regular level is connected;
  \item every interior critical point has index $1$, $2$, or $3$; and
  \item at each exceptional parameter there is either one critical-value
  crossing or one birth or death of an adjacent-index pair, necessarily of
  type $1$--$2$ or $2$--$3$.
\end{enumerate}
In particular, the movie contains no $0$--$1$ or $3$--$4$ birth or death.
\end{proposition}

\begin{proof}
Choose ordered Morse functions for the two handle decompositions.  Their
unique minima and maxima can be moved to fixed prescribed points by ambient
isotopies and put into the same quadratic form on small coordinate balls.
Removing the interiors of those balls leaves a cobordism from $S^3$ to
$S^3$, and the restrictions satisfy the boundary minimum/maximum condition
of \cite[Definition~2.1(i)]{GayWehrheimWoodward2012}.

Every regular level is connected.  Indeed, levels begin connected; attaching
index-$1$ or index-$2$ handles preserves connectedness.  An index-$3$ handle
that disconnected a level could not be repaired by any later index-$3$
handle, while the level immediately below the unique index-$4$ handle is the
connected sphere $S^3$.  Thus the two restrictions are Morse data in the
sense of \cite[Definition~2.1]{GayWehrheimWoodward2012}.

Apply the relative connected Cerf theorem
\cite[Theorem~3.6]{GayWehrheimWoodward2012}.  It gives a regular homotopy
satisfying the boundary minimum/maximum condition, with connected levels and
only finitely many special parameters, each a single critical-value crossing
or cusp.  We do not require the homotopy to fix a prescribed quadratic
collar pointwise; changes of the boundary collars are absorbed by the
ambient isotopies at the ends.  The
boundary minimum/maximum condition excludes interior critical points of
index $0$ and $4$ throughout the homotopy; this exclusion is made explicit
in the proof of the cited theorem.  Hence a cusp can create or cancel only an
index-$(1,2)$ or index-$(2,3)$ pair, proving all three assertions.
\end{proof}

\begin{proposition}[Special Kirby completeness]
\label{prop:exact-special-link-realization}
Let $D$ and $D'$ be prescribed special Kirby representatives of closed
connected oriented smooth four-manifolds $X$ and $Y$, and let
$f\colon X\to Y$ be an orientation-preserving diffeomorphism.  There is a
finite special-link realization of $f$ consisting of:
\begin{enumerate}[label=(\roman*),ref=\roman*]
  \item a finite sequence $D_0,\ldots,D_N$ of special Kirby
  representatives, with $D_0=D$ and $D_N=D'$;

  \item for every $j<N$, one of the eight unmarked directed events of
  \cref{def:kirby-event} from $D_j$ to $D_{j+1}$;

  \item a ball or standard cancelling-pair neighborhood supporting that
  event, outside which the special framed link is unchanged; and

  \item for every event, its canonical cellular continuation map, such that
  the composite map agrees under the cellular--singular comparison with the
  map induced by $f$.
\end{enumerate}
An index-$3$ slide is omitted from the finite sequence only after identifying
the two stages by the induced integral change of the displayed $C_3$ basis.
This is a silent equivalence of the data retained in
\cref{def:special-kirby-representative}: it preserves $h_3$ and
$\operatorname{im}(\partial_3)$, but is not the identity operation on the
full handle decomposition.  No category, character, graphical evaluation,
or scalar equality occurs in this statement.
\end{proposition}

\begin{proof}
Choose the neighborhoods of the unique index-$0$ and index-$4$ critical
points compatibly with $f$, remove their interiors, and pull the target
decomposition back by $f$.  This gives two relative handle decompositions of
the same cobordism from $S^3$ to $S^3$, containing only handles of indices
$1$, $2$, and $3$.  Use the relative fixed-extrema movie of
\cref{prop:relative-fixed-extrema-cerf}.

Gompf--Stipsicz's relative Kirby theorem
\cite[Theorem~5.5.3]{GompfStipsicz1999} converts this comparison into
isotopies in $\mathbb R^3$, handle slides, and creations or annihilations of
cancelling handle pairs.  Its proof is the Kirby-diagram form of the generic
handle argument in \cite[Theorem~4.2.12]{GompfStipsicz1999}.  Here the
incoming boundary is the standard $S^3$ with empty surgery diagram and its
parameterization has already been fixed, so the boundary-diagram Rolfsen
moves and insertions appearing in the general relative statement are absent.
The fixed-extrema condition leaves only $1$--$2$ and $2$--$3$ births and
deaths.

Section~5.4 of the same reference gives the required dotted-circle
translation: Figure~5.36 is the $2/1$ move, Figure~5.38 is the cancelling
$1$--$2$ pair, and Figure~5.39 is the $1/1$ slide; ordinary same-index
$2$-handle slides give the $2/2$ move.  Together with framed isotopy and the
unchanged $2$--$3$ cancellation picture, these are exactly the eight directed
events of \cref{def:kirby-event}.  Each displayed move is local, so its
standard ball or cancelling-pair neighborhood supplies (iii).

For each step take the canonical cellular continuation map.  A slide gives
the corresponding integral elementary basis change and a birth or death
gives the standard acyclic expansion or collapse, as in
\cref{lem:marked-handle-complex-naturality}.  Naturality of the
cellular--singular comparison identifies their composite with the map induced
by $f$.  Finally, an index-$3$ slide is an integral change of the $C_3$ basis
and may be suppressed by \cref{lem:c3-basis-independence}.  This proves all
four assertions.
\end{proof}

\begin{theorem}[Fixed-extrema marked Kirby--Cerf completeness]
\label{prop:marked-kirby-cerf}
Let $(D,\xi_D)$ and $(D',\xi_{D'})$ represent $(X,\alpha)$ and
$(Y,\alpha')$, respectively, and let $f\colon X\to Y$ be an
orientation-preserving diffeomorphism such that $f_*\alpha=\alpha'$.  A
finite sequence of the events in
\cref{def:marked-kirby-event} joins the two diagrams, and the composite of
their source-to-target handle-cohomology transports carries $\xi_D$ to
$\xi_{D'}$.
\end{theorem}

\begin{proof}
Pull the target handle decomposition back along the specified
diffeomorphism.  The comparison is then between two special handle
decompositions of the same oriented manifold.  Apply
\cref{prop:exact-special-link-realization} to obtain its finite unmarked
eight-event realization.

Write $F_j$ for the cellular continuation map of the $j$th event and
$T_j=(F_j^*)^{-1}$ for its source-to-target cohomology transport.  By
\cref{lem:marked-handle-complex-naturality}, the cellular comparison maps to
singular cohomology commute with each contravariant map $F_j^*$.  Hence
$T_j$ carries the handle class corresponding to a fixed singular-cohomology
class to the class corresponding to that same class in the next handle
decomposition.  Starting with $\xi_D$, define each intermediate marking by
$\xi_{j+1}:=T_j(\xi_j)$.  This turns the unmarked movie into a movie of marked
events without adding any topology or evaluation assumption.  Induction
shows that the composite of the $T_j$ carries $\xi_D$ to the marking of the
pulled-back target.

Undoing the pullback uses the naturality
identity
\[
  f^*\operatorname{PD}_Y(f_*\alpha)=\operatorname{PD}_X(\alpha)
\]
for an orientation-preserving diffeomorphism $f$.  Hence the composite
source-to-target transport carries $\xi_D$ to $\xi_{D'}$, as claimed.
\end{proof}

\begin{proposition}[Local movie normal forms]
\label{prop:local-movie-normal-forms}
The local forms required below satisfy the following conditions.
\begin{enumerate}[label=(\roman*),ref=\roman*]
  \item\label{item:slide-normal-form}
  After framed isotopy, every slide is supported in a $3$-ball and decomposes
  into a clean attachment, transport along the stationary component through
  finitely many local events, and a clean detachment.  In a $1/1$ slide the
  band is a legal dotted-handle slide: the target dotted sublink is again an
  unlink, and the source and target admit systems of disjoint spanning disks
  agreeing outside the supporting ball.

  \item\label{item:cancellation-normal-form}
  Every birth inserts an isolated standard cancelling pair.  Before a
  $1$--$2$ death, ordinary $2$-handles other than the cancelling meridian are
  slid off the cancelling $1$-handle; the resulting death removes an isolated
  standard pair.  The analogous $2$--$3$ death is put in its standard local
  form.  Thus a general cancellation is represented by allowed slides
  followed by an isolated cancellation, rather than evaluated as an
  isolated scalar without this reduction.

  \item\label{item:fixed-exterior}
  Outside the supporting ball, the framed tangle and every uninvolved
  component remain fixed.
\end{enumerate}
These are the standard local band-slide and cancelling-pair models of Kirby
calculus; compare \cite[Sections~5.1 and~5.4]{GompfStipsicz1999} and
\cite[Table~2]{BarenzBarrett2018}.  The topological assertion in
\cref{item:slide-normal-form} is only that the standard band admits clean
endpoint collars and a finite route with fixed exterior.  After categorical
decoration, the attach, transport, and detach morphisms are defined and
compared in \cref{subsec:kirby-invariance}; no graphical equality is imported
from this proposition.
\end{proposition}

\begin{proof}
Choose a band for the slide and compact collars at its endpoints.  A small
isotopy makes the collars clean.  Compactness of the stationary circle and
genericity leave only finitely many crossings, extrema, and framing twists
of the underlying tangle to pass.  Character and fusion coupons arise only
after categorical decoration and are handled separately by
\cref{lem:finite-coupon-transport}.  For a cancelling $1$--$2$ pair, slide
every other $2$-handle that
passes through the $1$-handle over the cancelling $2$-handle, with sign
chosen to remove that passage.  Finitely many passages are removed by
induction; this is a change of the $C_2$ basis and leaves an isolated
cancelling pair.  Handle cancellation then supplies the standard
birth/death model.  The $2$--$3$ case is the upside-down analogue.  These are
the local forms shown in
\cite[Table~2 and its caption]{BarenzBarrett2018}.
\end{proof}
