\section{Untitled Section}

We record here the numerical statements used in the introduction.  Let
$(\cC,\eta)$ be a semisimple premodular fusion category over $\mathbb C$
equipped with a monoidal natural automorphism of the identity.  Write
\[
  \widehat{\CY}_{\cC,\eta}(X,\alpha)
  := \operatorname{Dim}(\cC)^{-1}\CY_{\cC,\eta}(X,\alpha).
\]
For a simple object $i$, let $d_i$ be its quantum dimension, $\theta_i$ its
twist scalar, and $\eta_i\in\mathbb C^\times$ the scalar by which $\eta$ acts
on $i$.  Define the shifted Gauss sums
\[
  \tau_\eta^+(n)=\sum_{i\in\operatorname{Irr}(\cC)}d_i^2\theta_i\eta_i^n,
  \qquad
  \tau_\eta^-(n)=\sum_{i\in\operatorname{Irr}(\cC)}d_i^2\theta_i^{-1}\eta_i^n.
\]

\begin{proposition}[Marked diffeomorphism invariance]
\label{prop:diffeomorphism-invariance}
Let $X$ and $Y$ be closed oriented smooth four-manifolds, let
$f\colon X\to Y$ be an orientation-preserving diffeomorphism, and let
$\alpha\in H_2(X;\mathbb Z)$.  Then
\[
  \CY_{\cC,\eta}(X,\alpha)
  =\CY_{\cC,\eta}(Y,f_*\alpha).
\]
\end{proposition}

\begin{proposition}[Odd premodular formula]\label{prop:odd-formula}
Let $X$ be a closed, oriented, simply connected smooth four-manifold with odd
intersection form.  Choose an integral diagonalization
\[
  H_2(X;\mathbb Z)\cong
  \mathbb Z^p\oplus\mathbb Z^q,
  \qquad
  Q_X\cong p\langle 1\rangle\oplus q\langle-1\rangle,
\]
and write $\alpha=(a_1,\ldots,a_p;b_1,\ldots,b_q)$ in these coordinates.
Then
\[
  \widehat{\CY}_{\cC,\eta}(X,\alpha)
  =\prod_{r=1}^{p}\tau_\eta^+(a_r)
   \prod_{s=1}^{q}\tau_\eta^-(b_s).
\]
\end{proposition}

For the even formula, put $H=S^2\times S^2$.  For a marking $(a,b)$ in the
hyperbolic basis of $H_2(H;\mathbb Z)$, define
\[
  \mathcal H_{\cC,\eta}(a,b)
  :=\sum_{i,j\in\operatorname{Irr}(\cC)}
      d_i d_j\eta_i^a\eta_j^b
      \operatorname{Hopf}_{\cC}^{+}(i,j),
\]
where $\operatorname{Hopf}_{\cC}^{+}(i,j)$ is the positive Hopf-link scalar.
For a marked standard block $(B,\beta)$, set
\[
  V_{\cC,\eta}(B,\beta)=
  \begin{cases}
    \widehat{\CY}_{\cC,\eta}(K3,\beta),&B=K3,\\
    \widehat{\CY}_{\cC,\eta}(\overline{K3},\beta),&B=\overline{K3},\\
    \mathcal H_{\cC,\eta}(a,b),&B=H\text{ and }\beta=(a,b).
  \end{cases}
\]

\begin{proposition}[Even premodular formula]\label{prop:even-formula}
Let $X$ be a closed, oriented, simply connected smooth four-manifold with even
intersection form, and let $\alpha\in H_2(X;\mathbb Z)$.  Suppose that, for
some $s\geq0$, there is an integral intersection-form isometry of marked
lattices
\[
  \bigl(H_2(X\mathbin{\#}sH;\mathbb Z),Q_{X\#sH},\alpha\oplus0\bigr)
  \cong
  \bigoplus_{r=1}^{N}
    \bigl(H_2(B_r;\mathbb Z),Q_{B_r},\beta_r\bigr),
\]
where every $B_r$ is $K3$, $\overline{K3}$, or $H$.  Then
\[
  \widehat{\CY}_{\cC,\eta}(X,\alpha)
  =\frac{\displaystyle\prod_{r=1}^{N}
      V_{\cC,\eta}(B_r,\beta_r)}
    {\mathcal H_{\cC,\eta}(0,0)^s}.
\]
\end{proposition}

Now suppose that $\cC$ is modular.  Let
\[
  \tau_+(\cC)=\tau_\eta^+(0),
  \qquad
  \tau_-(\cC)=\tau_\eta^-(0),
\]
and let $g_\eta$ be the invertible simple object representing the character
defined by $\eta$.

\begin{proposition}[Modular closed formula]\label{prop:modular-formula}
For every closed, oriented, simply connected smooth four-manifold $X$ and
every $\alpha\in H_2(X;\mathbb Z)$,
\[
  \CY_{\cC,\eta}(X,\alpha)
  =\operatorname{Dim}(\cC)\,
    \tau_+(\cC)^{b_2^+(X)}
    \tau_-(\cC)^{b_2^-(X)}
    \theta_{g_\eta}^{-Q_X(\alpha,\alpha)}.
\]
\end{proposition}

\begin{corollary}[Simply connected no-go theorem]
\label{cor:no-simply-connected-exotic-detection}
Let $X$ and $Y$ be homeomorphic closed, oriented, simply connected smooth
four-manifolds, and let $f\colon X\to Y$ be an orientation-preserving
homeomorphism.  Then, for every $\alpha\in H_2(X;\mathbb Z)$,
\[
  \CY_{\cC,\eta}(X,\alpha)
  =\CY_{\cC,\eta}(Y,f_*\alpha).
\]
In particular, the family of marked invariants
$\CY_{\cC,\eta}(-,-)$ does not distinguish exotic smooth structures on a
simply connected topological four-manifold.
\end{corollary}

\begin{conjecture}[Non-simply-connected detection]
\label{conj:non-simply-connected-detection}
There exist a semisimple premodular fusion category $\cC$ over $\mathbb C$, a
monoidal natural automorphism $\eta$ of its identity functor, homeomorphic but
non-diffeomorphic closed oriented non-simply-connected smooth four-manifolds
$X$ and $Y$, an orientation-preserving homeomorphism $f\colon X\to Y$, and a
class $\alpha\in H_2(X;\mathbb Z)$ such that
\[
  \CY_{\cC,\eta}(X,\alpha)
  \ne\CY_{\cC,\eta}(Y,f_*\alpha).
\]
\end{conjecture}
